\begin{lemma}[One-cell sweep in an adapted van Kampen grid]
\label{lem:van-kampen-grid-cell-sweep}
\group{Fundamental Group}
\lean{HatcherLib.VanKampenSquareGrid.cellSweepStep_of_grid}
\leanok
Suppose a rectangular grid in a homotopy square is subordinate to the cover,
and at every grid vertex the labels of all incident rectangles occur among
three chosen cover members.  If all triple intersections of cover members
are path-connected, choose a path from the basepoint to each grid vertex in
the corresponding triple intersection.  For any cell, the word consisting
of its bottom and right edges, closed up by these vertex paths, is equivalent
to the word consisting of its left and top edges.  This replacement remains
valid inside an arbitrary prefix and suffix of a factorization word.
\end{lemma}

\begin{lemma}[Subordinate rectangular subdivision of a homotopy square]
\label{lem:van-kampen-rectangular-subdivision}
\group{Fundamental Group}
\lean{HatcherLib.exists_vanKampenRectangularSubdivision}
\leanok
Let $F:I\times I\to X$ be a homotopy of based paths and let
$\{A_\alpha\}_{\alpha\in J}$ be an open cover of $X$. There are finite monotone
subdivisions
\[
0=s_0\leq s_1\leq\cdots\leq s_m=1,
\qquad
0=t_0\leq t_1\leq\cdots\leq t_n=1
\]
such that for every $i<m$ and $j<n$ the image
$F([s_i,s_{i+1}]\times[t_j,t_{j+1}])$ is contained in some $A_\alpha$.
\end{lemma}
\begin{proof}
The inverse images $F^{-1}(A_\alpha)$ form an open cover of the compact metric
square $I\times I$. Choose a Lebesgue number $\delta>0$ for this cover, and choose
finite subdivisions of both copies of $I$ whose mesh is less than
$\delta/\sqrt 2$. Each resulting closed rectangle has diameter less than
$\delta$, hence lies in one $F^{-1}(A_\alpha)$, which is the required
containment after applying $F$.
\end{proof}
