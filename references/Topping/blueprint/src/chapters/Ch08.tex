\chapter{Perelman's \texorpdfstring{$\mathcal W$}{W} Entropy Functional}

\section{Perelman's \texorpdfstring{$\mathcal W$}{W} entropy functional}

For a metric $g$, a smooth function $f:\mathcal M\to\mathbb R$, and
$\tau>0$, set
\[
u=(4\pi\tau)^{-n/2}e^{-f},\qquad
\mathcal W(g,f,\tau)=\int_{\mathcal M}
\bigl[\tau(R+|\nabla f|^2)+f-n\bigr]u\,dV.
\]

\begin{definition}
\label{topping-ch8-compatible-arguments}
\source{topping:page-0086}
\dcref{ch8:1.1}
\group{perelman-w-entropy}
A triple $(g,f,\tau)$ is \emph{compatible} when
\[
\int_{\mathcal M}u\,dV
=\int_{\mathcal M}\frac{e^{-f}}{(4\pi\tau)^{n/2}}\,dV=1.
\]
\end{definition}

\begin{proposition}
\label{topping-ch8-scale-invariance}
\source{topping:page-0087}
\uses{topping-ch8-compatible-arguments}
\dcref{ch8:1.2}
\group{perelman-w-entropy}

The map $(g,f,\tau)\mapsto(\tau^{-1}g,f,1)$ preserves compatibility and
\[
\mathcal W(g,f,\tau)=\mathcal W(\tau^{-1}g,f,1).
\]
\end{proposition}

On $\mathbb R^n$, let
\[
d\mu=(2\pi)^{-n/2}e^{-|x|^2/2}\,dx,
\qquad \int_{\mathbb R^n}d\mu=1.
\]

\begin{theorem}
\label{topping-ch8-gross-log-sobolev}
\source{topping:page-0087}
\dcref{ch8:1.3}
\group{perelman-w-entropy}
If $v:\mathbb R^n\to\mathbb R$ is smooth with
$v,|\nabla v|\in L^2(d\mu)$, then
\[
\int v^2\ln|v|\,d\mu
\leq \int|\nabla v|^2\,d\mu
+\left(\int v^2\,d\mu\right)
 \ln\left(\int v^2\,d\mu\right)^{1/2}.
\]
\end{theorem}

\begin{remark}
\label{topping-ch8-gross-equality}
\source{topping:page-0087}
\dcref{ch8:1.4}
\group{perelman-w-entropy}
Equality in the Gross inequality holds if and only if $v$ is constant.
\end{remark}

\begin{remark}
\label{topping-ch8-gross-no-dimension-constant}
\source{topping:page-0087}
\dcref{ch8:1.5}
\group{perelman-w-entropy}
The Gross inequality has no dimension-dependent additive constant.
\end{remark}

\begin{remark}
\label{topping-ch8-gross-unit-mass}
\source{topping:page-0087}
\dcref{ch8:1.6}
\group{perelman-w-entropy}
If $\int v^2\,d\mu=1$, the Gross inequality reduces to
\[
\int v^2\ln|v|\,d\mu\leq\int|\nabla v|^2\,d\mu.
\]
\end{remark}

\begin{lemma}
\label{topping-ch8-euclidean-w-nonnegative}
\source{topping:page-0087,topping:page-0088,topping:page-0089}
\uses{topping-ch8-gross-log-sobolev, topping-ch8-scale-invariance}
\dcref{ch8:1.7}
\group{perelman-w-entropy}

If $g$ is the flat metric on $\mathcal M=\mathbb R^n$ and $(g,f,\tau)$ is
compatible, then
\[
\mathcal W(g,f,\tau)\geq0,
\]
with equality if and only if
$f(x)=|x|^2/(4\tau)$.
\end{lemma}

\begin{proof}
\sketch By scale invariance it is enough to take $\tau=1/2$. Define
$v=e^{|x|^2/4-f/2}$. Compatibility gives
$\int v^2\,d\mu=1$, so the unit-mass Gross inequality applies. Its left
side is
\[
\int\left(\frac{|x|^2}{4}-\frac f2\right)
\frac{e^{-f}}{(2\pi)^{n/2}}\,dx.
\]
Using $\nabla\!\cdot x=n$ and integration by parts,
\[
-\int\frac{x\cdot\nabla f}{2}\frac{e^{-f}}{(2\pi)^{n/2}}\,dx
=-\frac n2\int\frac{e^{-f}}{(2\pi)^{n/2}}\,dx,
\]
and hence
\[
\int|\nabla v|^2\,d\mu
=\int\left(\frac{|x|^2}{4}-\frac n2+\frac{|\nabla f|^2}{4}\right)
\frac{e^{-f}}{(2\pi)^{n/2}}\,dx.
\]
The resulting inequality is $\mathcal W(g,f,1/2)\geq0$. Equality in the
Gross inequality gives $f(x)=|x|^2/2$ at $\tau=1/2$, and rescaling gives the
stated equality case.
\end{proof}

\begin{lemma}
\label{topping-ch8-minimiser-exists}
\source{topping:page-0089,topping:page-0090,topping:page-0091}
\uses{topping-ch8-scale-invariance}
\dcref{ch8:1.8}
\group{perelman-w-entropy}

For every smooth metric $g$ on a closed manifold $\mathcal M$ and every
$\tau>0$, the infimum of $\mathcal W(g,f,\tau)$ over compatible smooth $f$
is attained by a smooth compatible function.
\end{lemma}
\begin{proof}
\sketch Set $\phi=e^{-f/2}$. Use Proposition~\ref{topping-ch8-scale-invariance} to scale simultaneously
so that $\tau=1/(4\pi)$, and impose $\int\phi^2\,dV=1$. For a constant
depending only on $(g,n)$, bounded scalar curvature gives
\[
\int\left[\frac{R}{4\pi}\phi^2-n\phi^2\right]dV\geq-C(g,n).
\]
Set $d\mu=\phi^2dV$, $\sigma=\phi^{4/(n-2)}$, and $G(y)=\ln y$. Since
$G$ is concave and $d\mu$ has unit mass, Jensen's inequality and the Sobolev
inequality give
\[
\int\phi^2\ln\phi\,dV
\leq\frac{n-2}{4}\ln\left(\int\phi^{2+4/(n-2)}dV\right)
\leq\frac{1}{4\pi}\int|\nabla\phi|^2dV+C(g,n).
\]
Therefore
\[
\mathcal W(g,\phi,1/(4\pi))
\geq\frac{1}{2\pi}\int|\nabla\phi|^2dV-C(g,n).
\]
Thus a minimizing sequence is bounded in $W^{1,2}$; the direct method gives a
minimizer, and the positivity argument gives a smooth minimizing $f$.
\end{proof}

Define
\[
\mu(g,\tau):=\inf_f\mathcal W(g,f,\tau),
\qquad
\nu(g,\tau_0):=\inf_{\tau\in(0,\tau_0]}\mu(g,\tau).
\]
For every finite $\tau_0>0$, $\mu(g,\tau)$ is bounded below on
$(0,\tau_0]$, and consequently $\nu(g,\tau_0)>-\infty$.

With $\phi=e^{-f/2}$, allowing $\phi:\mathcal M\to[0,\infty)$, one has
\[
\mathcal W(g,\phi,\tau)
=(4\pi\tau)^{-n/2}\int\left[
\tau\bigl(4|\nabla\phi|^2+R\phi^2\bigr)-2\phi^2\ln\phi-n\phi^2
\right]dV,
\]
and the compatibility constraint is
\[
(4\pi\tau)^{-n/2}\int\phi^2\,dV=1.
\]
Approximation gives
\[
\inf_f\mathcal W(g,f,\tau)=\inf_\phi\mathcal W(g,\phi,\tau),
\]
and a minimizing weakly positive $\phi$ is strictly positive, yielding the
minimizing $f$ in Lemma~\ref{topping-ch8-minimiser-exists}. The boundedness of $\mu$ on finite $\tau$
intervals is an additional analytic input from [31].
\subsection{Monotonicity of $\mathcal W$}

\begin{proposition}
\label{topping-ch8-monotonicity}
\source{topping:page-0092}
\dcref{ch8:2.1}
\group{perelman-w-entropy}
If $\mathcal M$ is closed and
\[
\frac{\partial g}{\partial t}=-2\Ric,\qquad
\frac{d\tau}{dt}=-1,\qquad
\frac{\partial f}{\partial t}=-\Delta f+|\nabla f|^2-R+\frac{n}{2\tau},
\]
then
\[
\frac{d}{dt}\mathcal W(g,f,\tau)
=2\tau\int\left|\Ric+\Hess(f)-\frac{g}{2\tau}\right|^2dV\geq0.
\]
\end{proposition}

\begin{remark}
\label{topping-ch8-compatibility-preserved}
\source{topping:page-0092}
\dcref{ch8:2.2}
\group{perelman-w-entropy}
Under this coupled evolution, $u$ satisfies $\Box^*u=0$, so compatibility is
preserved.
\end{remark}

\begin{remark}
\label{topping-ch8-shrinking-soliton}
\source{topping:page-0092}
\dcref{ch8:2.3}
\group{perelman-w-entropy}
If the derivative in Proposition~\ref{topping-ch8-monotonicity} is not strictly positive, then
\[
\Ric+\Hess(f)-\frac{g}{2\tau}=0,
\]
and the flow is a shrinking gradient soliton.
\end{remark}

\begin{remark}
\label{topping-ch8-existence-of-coupled-system}
\source{topping:page-0093}
\dcref{ch8:2.5}
\group{perelman-w-entropy}
Given a Ricci flow on $[0,T]$, one may set $\tau=T+C-t$ and solve the linear
equation for $u$ with prescribed final data, then recover $f$.
\end{remark}

\begin{proposition}
\label{topping-ch8-local-monotonicity}
\source{topping:page-0093,topping:page-0094,topping:page-0095}
\uses{topping-ch8-monotonicity, topping-ch8-compatibility-preserved}
\dcref{ch8:2.6}
\group{perelman-w-entropy}

Under the evolution in Proposition~\ref{topping-ch8-monotonicity}, define
\[
v=\bigl[\tau(2\Delta f-|\nabla f|^2+R)+f-n\bigr]u.
\]
Then
\[
\Box^*v=-2\tau\left|\Ric+\Hess(f)-\frac{g}{2\tau}\right|^2u.
\]
\end{proposition}

\begin{proof}
\sketch Write $v=(v/u)u$. The product rule and $\Box^*u=0$ give
\[
u^{-1}\Box^*v=-\left(\partial_t+\Delta\right)(v/u)
-2u^{-1}\left\langle\nabla(v/u),\nabla u\right\rangle.
\]
For $A=2\Delta f-|\nabla f|^2+R$, the coupled evolution equations yield
\[
\left(\partial_t+\Delta\right)A
=4\langle\Ric,\Hess(f)\rangle+\Delta|\nabla f|^2
-2\Ric(\nabla f,\nabla f)
-2\langle\nabla f,\nabla(-\Delta f+|\nabla f|^2-R)\rangle+2|\Ric|^2,
\]
while $\nabla u=-u\nabla f$. Substitution and cancellation give
\[
u^{-1}\Box^*v
=-\frac{n}{2\tau}+2\langle\Ric+\Hess(f),g\rangle
-2\tau|\Ric+\Hess(f)|^2
=-2\tau\left|\Ric+\Hess(f)-\frac{g}{2\tau}\right|^2.
\]
\end{proof}

\begin{remark}
\label{topping-ch8-proposition-implies-monotonicity}
\source{topping:page-0093}
\dcref{ch8:2.7}
\group{perelman-w-entropy}
Since $\mathcal W=\int_{\mathcal M}v\,dV$, integrating the local formula
gives Proposition~\ref{topping-ch8-monotonicity}.
\end{remark}

\subsection{No local volume collapse where curvature is controlled}

For a complete $(\mathcal M,g)$, $p\in\mathcal M$, and $s>0$, write
\[
\mathcal V(p,s)=\mathcal V_g(p,s)=\operatorname{Vol}_g(B_g(p,s)),
\qquad
\mathcal K(p,r)=\frac{\mathcal V(p,r)}{r^n}.
\]

\begin{theorem}
\label{topping-ch8-no-collapsing}
\source{topping:page-0096}
\uses{topping-ch8-mu-ball-estimate}
\dcref{ch8:3.1}
\group{perelman-w-entropy}

Suppose $g(t)$ is a Ricci flow on a closed $\mathcal M$ for $t\in[0,T]$.
With all quantities computed using $g(T)$, if $p\in\mathcal M$ and $r>0$ is
sufficiently small that $|R|\leq r^{-2}$ on $B(p,r)$, then
\[
\mathcal K(p,r)>\xi
\]
for some $\xi>0$ depending on $n$, $g(0)$, and upper bounds for $r$ and $T$.
\end{theorem}

\begin{proof}
\sketch Let $r_0,T_0$ bound $r,T$. Theorem~\ref{topping-ch8-mu-ball-estimate} gives
$\gamma=\gamma(n,g(0),T_0,r_0)$ such that for $s\in(0,r_0]$,
\[
\gamma\leq\frac{\mathcal V(p,s)}{\mathcal V(p,s/2)}
+\frac{s^2}{\mathcal V(p,s/2)}\int_{B(p,s)}|R|\,dV+\ln\mathcal K(p,s).
\]
For $s\leq r$, the curvature hypothesis gives
\[
\gamma\leq2\frac{\mathcal V(p,s)}{\mathcal V(p,s/2)}+\ln\mathcal K(p,s).
\]
Let $\omega_n$ be the Euclidean unit-ball volume and set
\[
\xi=\min\{\omega_n/2,e^{\gamma-2^{n+1}}\}>0.
\]
If $\mathcal K(p,s)\leq\xi$ for $s\leq r$, the preceding inequality implies
\[
2^n\leq\frac{\mathcal V(p,s)}{\mathcal V(p,s/2)}
=2^n\frac{\mathcal K(p,s)}{\mathcal K(p,s/2)},
\]
so $\mathcal K(p,s/2)\leq\mathcal K(p,s)\leq\xi$. Iteration would contradict
$\mathcal K(p,s)\to\omega_n$ as $s\downarrow0$; hence
$\mathcal K(p,r)>\xi$.
\end{proof}

\begin{theorem}
\label{topping-ch8-mu-ball-estimate}
\source{topping:page-0096}
\uses{topping-ch8-minimiser-exists, topping-ch8-monotonicity}
\dcref{ch8:3.4}
\group{perelman-w-entropy}

Suppose $g(t)$ is a Ricci flow on a closed $\mathcal M$ for $t\in[0,T]$.
Compute $\mathcal V$, $R$, and balls using $g(T)$. For $p\in\mathcal M$ and
$r>0$,
\[
\gamma\leq\frac{\mathcal V(p,r)}{\mathcal V(p,r/2)}
+\frac{r^2}{\mathcal V(p,r/2)}\int_{B(p,r)}|R|\,dV
+\ln\mathcal K(p,r),
\]
where $\gamma\in\mathbb R$ depends on $n$, $g(0)$, and upper bounds for $r$
and $T$.
\end{theorem}

\begin{proof}
\sketch Apply Lemma~\ref{topping-ch8-mu-ball-bound} with $g=g(T)$ and $\lambda=1/36$. This gives
\[
\mu\left(g(T),\frac{r^2}{36}\right)
\leq\frac{\mathcal V(p,r)}{\mathcal V(p,r/2)}
+\frac{r^2}{\mathcal V(p,r/2)}\int_{B(p,r)}|R|\,dV
+\ln\mathcal K(p,r)-\frac n2\ln\frac\pi9-n.
\]
Choose a minimizing compatible $f_T$ at $\tau=r^2/36$ and solve the coupled
system backward with $\tau=T+r^2/36-t$ and final value $f_T$. Compatibility is
preserved, and monotonicity gives
\[
\mu\left(g(0),T+\frac{r^2}{36}\right)
\leq\mathcal W\left(g(0),f(0),T+\frac{r^2}{36}\right)
\leq\mu\left(g(T),\frac{r^2}{36}\right).
\]
The lower bound defining $\nu(g(0),T+r_0^2/36)$ absorbs the constants and
yields the stated $\gamma$.
\end{proof}

\begin{lemma}
\label{topping-ch8-mu-ball-bound}
\source{topping:page-0096,topping:page-0097,topping:page-0098}
\uses{topping-ch8-minimiser-exists}
\dcref{ch8:3.5}
\group{perelman-w-entropy}

For any smooth metric $g$ on a closed $\mathcal M$, $r>0$, $p\in\mathcal M$,
and $\lambda>0$,
\[
\mu(g,\lambda r^2)\leq36\lambda
\frac{\mathcal V(p,r)-\mathcal V(p,r/2)}{\mathcal V(p,r/2)}
+\frac{\lambda r^2}{\mathcal V(p,r/2)}\int_{B(p,r)}|R|\,dV
+\ln\left[\frac{\mathcal V(p,r)}{(4\pi\lambda r^2)^{n/2}}\right]-n.
\]
\end{lemma}

\begin{proof}
\sketch Put $\tau=\lambda r^2$ and $\phi=e^{-f/2}$ in the expression for
$\mathcal W$; approximation permits nonnegative $\phi$ and preserves the
infimum. Choose
\[
\phi(x)=e^{-c/2}\psi(d(x,p)/r),
\]
where $\psi:[0,\infty)\to[0,1]$ is supported in $[0,1]$, equals $1$ on
$[0,1/2]$, and satisfies $|\psi'|\leq3$. The constraint implies
\[
(4\pi\lambda r^2)^{-n/2}\mathcal V(p,r/2)
\leq e^c\leq
(4\pi\lambda r^2)^{-n/2}\mathcal V(p,r).
\]
The gradient is supported in $B(p,r)\setminus B(p,r/2)$ and gives
\[
(4\pi\lambda r^2)^{-n/2}\int4\lambda r^2|\nabla\phi|^2dV
\leq36\lambda\frac{\mathcal V(p,r)-\mathcal V(p,r/2)}{\mathcal V(p,r/2)}.
\]
The scalar-curvature term is bounded by
\[
\frac{\lambda r^2}{\mathcal V(p,r/2)}\int_{B(p,r)}|R|\,dV.
\]
For the logarithmic term, use the concavity of $G(y)=-y\ln y$ with
$d\mu=(4\pi\lambda r^2)^{-n/2}dV$ and $\sigma=\phi^2$; Jensen's inequality
and the constraint give
\[
(4\pi\lambda r^2)^{-n/2}\int-2\phi^2\ln\phi\,dV
\leq\ln\left[\frac{\mathcal V(p,r)}{(4\pi\lambda r^2)^{n/2}}\right].
\]
The final term is $-n$ by compatibility. Adding the four estimates proves
the bound.
\end{proof}

\subsection{Volume ratio bounds imply injectivity radius bounds}

For a Riemannian manifold $(\mathcal M,g)$, write
\[
\operatorname{inj}(\mathcal M)=\operatorname{inj}(\mathcal M,g)
:=\inf_{p\in\mathcal M}\operatorname{inj}(\mathcal M,g,p).
\]

\begin{lemma}
\label{topping-ch8-volume-ratio-inj-estimate}
\source{topping:page-0101,topping:page-0102,topping:page-0103}
\uses{topping-ch8-no-collapsing}
\dcref{ch8:4.1}
\group{perelman-w-entropy}

There are a universal $\bar r>0$ and an $n$-dependent $\eta>0$ such that,
if $(\mathcal M^n,g)$ is closed and $|\operatorname{Rm}|\leq1$, then some
$p\in\mathcal M$ satisfies
\[
\frac{\mathcal V(p,r)}{r^n}\leq\frac\eta r\operatorname{inj}(\mathcal M)
\]
for every $r\in(0,\bar r]$.
\end{lemma}

\begin{proof}
\sketch Reduce $\bar r$ as needed. If $\operatorname{inj}(\mathcal M)\geq\pi$,
Bishop comparison under the sectional-curvature lower bound $-1$ gives an
$n$-dependent upper bound $C$ for the volume ratio on $r\leq1$, so the result
holds with $\eta=C/\pi$. Otherwise Klingenberg's lemma supplies a closed
geodesic loop $\gamma$ of length $2\operatorname{inj}(\mathcal M)$; choose
$p\in\gamma$.

For sufficiently small $r$, the normal disk bundle $N^r\gamma$ exponentiates
onto the $r$-neighborhood $\mathcal N^r\gamma$ and Jacobi-field estimates give
$u^*g\leq4G$, where $G$ is the natural flat metric on $N^r\gamma$. Therefore
\[
\operatorname{Vol}_g(\mathcal N^r\gamma)
\leq2^n\operatorname{Length}(\gamma)\omega_{n-1}r^{n-1}.
\]
Since $B_g(p,r)\subset\mathcal N^r\gamma$,
\[
\frac{\mathcal V(p,r)}{r^n}
\leq2^{n+1}\omega_{n-1}\frac{\operatorname{inj}(\mathcal M)}r.
\]
\end{proof}

\subsection{Blowing up at singularities II}

Let $g(t)$ be a Ricci flow on a closed $\mathcal M$ with maximal finite time
$T$. Choose $p_i\in\mathcal M$ and $t_i\uparrow T$ so that
\[
|\Rm|(p_i,t_i)=\sup_{x\in\mathcal M,\,t\in[0,t_i]}|\Rm|(x,t)\to\infty,
\]
and set
\[
g_i(t)=|\Rm|(p_i,t_i)
g\left(t_i+\frac{t}{|\Rm|(p_i,t_i)}\right).
\]
For $a<0$ and some $b=b(n)>0$, the curvature of $g_i(t)$ is controlled on
$(a,b)$; in particular, for sufficiently large $i$,
\[
\sup_{x\in\mathcal M,\,t\in(a,b)}|R(g_i(t))|\leq M<\infty,
\]
where $M$ depends only on $n$. Thus $|R(g_i(0))|\leq r^{-2}$ for
$r<M^{-1/2}$, and in the original scale this holds when
$0<r\leq(M|\Rm|(p_i,t_i))^{-1/2}$.

\begin{theorem}
\label{topping-ch8-blow-up-singularities}
\source{topping:page-0104}
\uses{topping-ch8-no-collapsing, topping-ch8-volume-ratio-inj-estimate, topping-ch7-compactness-ricci-flows}
\dcref{ch8:5.1}
\group{perelman-w-entropy}

Suppose that $\mathcal M$ is closed and $g(t)$ is a Ricci flow on a maximal
interval $[0,T)$ with $T<\infty$. There are sequences $p_i\in\mathcal M$ and
$t_i\uparrow T$ with
\[
|\Rm|(p_i,t_i)=\sup_{x\in\mathcal M,\,t\in[0,t_i]}|\Rm|(x,t)\to\infty,
\]
such that, for
\[
g_i(t)=|\Rm|(p_i,t_i)
g\left(t_i+\frac{t}{|\Rm|(p_i,t_i)}\right),
\]
there exist $b=b(n)>0$, a complete Ricci flow $(\mathcal N,\hat g(t))$ for
$t\in(-\infty,b)$, and $p_\infty\in\mathcal N$ with
\[
(\mathcal M,g_i(t),p_i)\longrightarrow(\mathcal N,\hat g(t),p_\infty).
\]
Moreover,
\[
|\Rm(\hat g(0))|(p_\infty)=1,
\qquad |\Rm(\hat g(t))|\leq1\quad(t\leq0).
\]
\end{theorem}

\begin{proof}
\sketch Theorem~\ref{topping-ch8-no-collapsing} gives, at time $t_i$, a uniform lower bound
$\mathcal V(p,r)/r^n>\xi>0$ for all $p$ and
$0<r\leq(M|\Rm|(p_i,t_i))^{-1/2}$, for large $i$. After rescaling, the same
bound holds for $g_i(0)$ on $r\leq M^{-1/2}$. Taking
$r=\min\{M^{-1/2},\bar r\}$ in Lemma~\ref{topping-ch8-volume-ratio-inj-estimate} yields
\[
\operatorname{inj}(\mathcal M,g_i(0))
\geq\frac r\eta\frac{\mathcal V(p,r)}{r^n}
>\frac{r\xi}{\eta}>0.
\]
The bound depends only on $n$, $g(0)$, and $T$. The curvature control and this
injectivity-radius bound allow Theorem~\ref{topping-ch7-compactness-ricci-flows} to extract a pointed smooth limit
on $(-\infty,b)$; the normalization at $(p_i,t_i)$ gives
$|\Rm(\hat g(0))|(p_\infty)=1$ and the maximal-point choice gives
$|\Rm(\hat g(t))|\leq1$ for $t\leq0$.
\end{proof}
