\chapter{Connected Sum}

\section{Connected Sum}

\begin{definition}[Connected sum]
For oriented manifolds $\mathcal{A}$ and $\mathcal{B}$ of the same dimension,
the connected sum $\mathcal{A}\#\mathcal{B}$ is obtained by removing a small
disc or ball from each manifold and gluing the resulting boundary components
compatibly with the orientations.
\end{definition}
