\begin{definition}[Standard initial metric]
\label{def:standard-initial-metric}
\uses{def:sectional-curvature}
\label{defn:stinmetr}

A \emph{standard initial metric} is a complete $SO(3)$-invariant metric
$g_0$ on $\Ar^3$ with nonnegative sectional curvature, equal outside a
compact ball to $(S^2,h)\times(\Ar^+,ds^2)$, where $h$ has scalar
curvature $1$, and of constant sectional curvature $1/4$ near the origin
$p$, called its \emph{tip}.
\end{definition}

\begin{lemma}[Existence of a standard initial metric]
\label{lem:existence-standard-initial-metric}
\uses{def:standard-initial-metric}
\label{stdinit}

There is a standard initial metric.
\end{lemma}
\begin{proof}
\uses{def:standard-initial-metric}
\sketch Choose a smooth positive concave $f$ on $(0,\infty)$, equal to
$\sqrt{4s-s^2}$ near $0$ and to $\sqrt2$ for large $s$, and rotate its
graph to obtain
\[
 \Sigma(f)=\{(x_0,x_1,x_2,x_3):x_0\ge0,\ x_1^2+x_2^2+x_3^2=f(x_0)^2\}.
\]
The induced metric is smooth, complete, and cylindrical at infinity. Its
orbit and radial sectional curvatures are respectively
\[
 \frac1{f^2(1+(f')^2)},\qquad\frac{-f''}{f(1+(f')^2)^2},
\]
so it has the required curvature and tip. One may take
$f(s)=\int_0^s(2-\sigma)(4\sigma-\sigma^2)^{-1/2}
\lambda(\sigma-s_0)\,d\sigma$, choosing $s_0\in(1/2,3/2)$ so that
$\int_0^2(2-s)(4s-s^2)^{-1/2}\lambda(s-s_0)\,ds=\sqrt2$.
\end{proof}

\begin{lemma}[Asymptotic and curvature bounds for the standard initial metric]
\label{lem:standard-initial-metric-asymptotic-bounds}
\uses{lem:existence-standard-initial-metric}
\label{A_0}

There are $A_0,K,D<\infty$ such that $(\Ar^3\setminus B(0,A_0),g(0))$
is the round scalar-curvature-$1$ cylinder, $\Vol B_{g(0)}(0,A_0)\le K$,
and $D^{-1}\le R_{g(0)}\le D$.
\end{lemma}

Fix a standard initial metric $g_0$.

\begin{definition}[Partial and standard Ricci flow]
\label{def:standard-ricci-flow}
\uses{def:standard-initial-metric, def:ricci-flow}

A \emph{partial standard Ricci flow} is a Ricci flow $(\Ar^3,g(t))$,
$0\le t<T$, with $g(0)=g_0$ and curvature locally bounded in time. It is
\emph{standard} if it has no such extension to a larger interval.
\end{definition}

\begin{theorem}[Existence, uniqueness, and properties of the standard solution]
\label{thm:standard-solution-existence-uniqueness}
\uses{def:standard-ricci-flow, def:epsilon-neck}
\label{stdsoln}

A standard Ricci flow exists. Every standard flow $(\Ar^3,g(t))$, $0\le t<T$,
is unique, has $T=1$, is complete with strictly positive curvature for
$0<t<1$, is $SO(3)$-invariant, is asymptotic on every $[0,t_0]$, $t_0<1$,
to $(S^2,h(t))\times\Ar$ with $R_{h(t)}=(1-t)^{-1}$, and is
$\kappa$-noncollapsed below some scale $r>0$.
\end{theorem}
\begin{proof}
\sketch Double increasingly large cylindrical balls in $(\Ar^3,g_0)$.
The compact flows have a uniform short-time curvature bound by the scalar
maximum principle and retain nonnegative curvature. The geometric limit is
a bounded-curvature partial flow; a maximal locally bounded extension is
standard. The later arguments prove the asserted geometric properties and
uniqueness using the Ricci--DeTurck construction.
\end{proof}

\begin{lemma}[Completeness and positive curvature of the standard flow]
\label{lem:standard-flow-complete-positive-curvature}
\uses{def:standard-ricci-flow}
\label{third}

For every $t_0<T$, $(\Ar^3,g(t_0))$ is complete and has positive curvature.
\end{lemma}
\begin{proof}
\uses{prop:forward-difference-maximum,lem:curvature-evolution-frame,nullspace}
\sketch Bounded curvature on $[0,t_0]$ gives $g(0)\le Cg(t_0)$, hence
completeness. A proper barrier $\varphi$ with
$(\partial_t-\triangle)\varphi\ge C\varphi$ applies the tensor maximum
principle to $\mathcal T+\varphi g$, proving nonnegative curvature;
the nullspace principle gives strict positivity for $t_0>0$.
\end{proof}

\begin{proposition}[Asymptotic convergence to the standard evolving cylinder]
\label{prop:standard-flow-asymptotic-cylinder}
\uses{def:standard-ricci-flow, ex:shrinking-round-cylinder}
\label{item5}

For $T'<T$ and $\epsilon>0$, outside a compact ball the flow on
$[0,T']$ is $\epsilon$-close in $C^{[1/\epsilon]}$ to
$(S^2,h(t))\times(\Ar^+,ds^2)$.
\end{proposition}
\begin{proof}
\sketch Take pointed limits along the initial cylindrical end. Derivative
bounds and splitting give a product flow; positivity identifies its compact
factor with the shrinking round $S^2$.
\end{proof}

\begin{corollary}[Maximal time is at most one]
\label{cor:standard-flow-time-bound-one}
\uses{prop:standard-flow-asymptotic-cylinder}
\label{1}

$T\le1$.
\end{corollary}
\begin{proof}
\uses{prop:standard-flow-asymptotic-cylinder}
\sketch If $T>1$, the cylindrical asymptotic flow would extend past its
singular time $1$.
\end{proof}

\section{Rotational symmetry}

For $\partial_tX=\triangle X+\Ric(X,\cdot)^*$, write $X^*=g(X,\cdot)$:
\begin{equation}\label{eq1}\partial_tX=\triangle X+\Ric(X,\cdot)^*,\end{equation}
\begin{equation}\label{eq.3}\partial_tX^i=(\Delta X)^i+\Ric^i_kX^k.\end{equation}

\begin{lemma}[Evolution of $V=\nabla X^*$]
\label{lem:v-tensor-evolution}
\uses{def:standard-ricci-flow}
\label{12.9}

For $V_{ij}=(\nabla_iX)_j$,
\begin{equation}\label{eq.4}
 \partial_tV=\triangle V-\left(2{{R_k}^{rl}}_jV_{rl}+\Ric_k^lV_{lj}
 +\Ric_j^lV_{kl}\right)dx^k\otimes dx^j.
\end{equation}
\end{lemma}
\begin{proof}
\sketch Differentiate $V=\nabla X^*$ in normal coordinates, commute
covariant derivatives, and use the contracted Bianchi identity; the
intermediate identity is
\begin{equation}\label{eqn1}
 \partial_tV=(-\nabla^l\Ric_{kj}+\nabla_k\Ric_j^l+\nabla_j\Ric_k^l)X_l^*dx^k\otimes dx^j
 +\nabla(\Delta X^*-\Ric(X,\cdot)).
\end{equation}
\end{proof}

\begin{remark}[Covariant derivative and curvature acting on one-forms]
\label{rem:covariant-derivative-one-forms}
\uses{def:riemann-curvature-tensor}

For one-forms, $\nabla_{\partial_r}dx^k=-\Gamma^k_{rl}dx^l$ and
$\Rm(\partial_i,\partial_j)dx^k=-{{R_{ij}}^k}_l dx^l$; consequently
$\Delta X_i-\Ric_{ik}X^k=-\Delta_dX_i$.
\end{remark}

\begin{claim}[Commuting covariant derivatives and curvature]
\label{claim:commuting-covariant-derivatives}
\uses{def:riemann-curvature-tensor}

For each tensor $\phi$,
\[\nabla(\nabla_r\phi)=\nabla_r(\nabla\phi)+dx^k\otimes
\Rm(\partial_k,\partial_r)\phi-\nabla_r(dx^l)\otimes\nabla_l\phi.\]
\end{claim}

Set $h_{ij}=V_{ij}+V_{ji}$. Then
\begin{equation}\label{eq.5}\partial_th_{ij}=\Delta_Lh_{ij},\end{equation}
where $\Delta_Lh_{ij}=\Delta h_{ij}+2{{R_i}^{kl}}_jh_{kl}-\Ric_i^kh_{kj}-\Ric_j^kh_{ki}$, and
\begin{align}
 (\partial_t-\Delta)|h|^2&=-2|\nabla h|^2+4R^{ijkl}h_{jk}h_{il},\label{eq.6}\\
 \partial_t|h|^2&\le\Delta|h|^2-2|\nabla h|^2+C|h|^2.\label{eq.7}
\end{align}

\begin{corollary}[Standard solution is rotationally symmetric]
\label{cor:standard-solution-rotationally-symmetric}
\uses{def:standard-ricci-flow}

Every $g(t)$ is invariant under the standard $SO(3)$ action.
\end{corollary}
\begin{proof}
\sketch Evolve the initial rotational Killing fields by \eqref{eq1}.
The maximum principle applied to \eqref{eq.7}, using cylindrical decay,
gives $h=0$. The fields are stationary by \eqref{eq.3}, so the initial
$SO(3)$ action remains isometric.
\end{proof}

\begin{proposition}[Non-collapsing of the standard flow]
\label{prop:standard-flow-kappa-noncollapsed}
\uses{def:standard-ricci-flow}
\label{stdkappa}

For all sufficiently small $r>0$, the standard flow is
$\kappa$-noncollapsed on scales at most $r$ for some $\kappa>0$.
\end{proposition}
\begin{proof}
  \uses{prop:forward-difference-maximum, n/2, THM, def:standard-ricci-flow, prop:standard-flow-asymptotic-cylinder}

\sketch With $C=\max(2,\sup R(\cdot,0))$, the maximum principle gives
the following bound. The bounded initial interval has uniform volume ratios;
later times have a uniform reduced-volume lower bound, so the noncollapsing
theorem applies.
\end{proof}
\begin{claim}[Curvature bound along the standard flow]
\label{claim:standard-flow-curvature-bound}
\uses{def:standard-ricci-flow}

For $t_0<T$, $t_0<C^{-1}$, and $0\le t\le t_0$,
\[R(x,t)\le\frac C{1-Ct}.\]
\end{claim}
\begin{proof}
\uses{prop:standard-flow-asymptotic-cylinder}
\sketch At infinity $R<2/(1-t)$. On the compact set where a maximum can
exceed this value, $\partial_tR\le R^2$; comparison with $C/(1-Ct)$ proves
the claim.
\end{proof}

\section{Uniqueness}\label{12.4}
The Ricci--DeTurck reduction follows \cite{DeTurck}\cite{LuTian}; the general
bounded-curvature uniqueness result is \cite{ChenZhu}. The barrier and
compactness ingredients used above follow \cite{Hamiltonlimits}\cite{Shi1},
and the parabolic iteration uses \cite{LiTam}\cite{LiTam}\cite{LSU}\cite{LSU}.
The alternative uniqueness observation is \cite{ChenZhu}; the symmetry
argument is due to \cite{ChowKnopf}.

For a Ricci flow $g(t)$ and $\psi_{t_0}=\Id$, use the harmonic map flow
\begin{equation}\label{eq2.1}\partial_t\psi_t=\Delta_{g(t),g(t_0)}\psi_t,\end{equation}
whose coordinate form is
\begin{equation}\label{eq2.2}
(\partial_t-\Delta_{g(t)})\psi^\alpha=g^{ij}\Gamma^\alpha_{\beta\gamma}(\psi)\partial_i\psi^\beta\partial_j\psi^\gamma.
\end{equation}
For $\hat g=(\psi_t^{-1})^*g(t)$,
\begin{equation}\label{RDfloweqn}
\partial_t\hat g_{ij}=-2\widehat{\Ric}_{ij}+\hat\nabla_iW_j+\hat\nabla_jW_i,
\end{equation}
equivalently the strictly parabolic Ricci--DeTurck equation
\begin{equation}\label{eqRcDeTurck}
\partial_t\hat g_{ij}=\hat g^{kl}\nabla_k\nabla_l\hat g_{ij}+Q_{ij}(\hat g,\nabla\hat g).
\end{equation}

\begin{lemma}[Harmonic map flow gives a Ricci-DeTurck flow]
\label{lem:harmonic-map-flow-ricci-deturck}
\uses{def:standard-ricci-flow}
\label{12.7}

If $g(t)$ solves Ricci flow and $\psi_t$ solves \eqref{eq2.1} and is a
diffeomorphism, then $(\psi_t^{-1})^*g(t)$ solves \eqref{RDfloweqn}.
\end{lemma}
\begin{proof}
\sketch Differentiate the pullback and substitute the tension field of
$\psi_t$; its Lie derivative is the DeTurck correction.
\end{proof}

\begin{corollary}[Ricci-DeTurck uniqueness implies Ricci flow uniqueness]
\label{cor:ricci-deturck-uniqueness-implies-ricci-uniqueness}
\uses{lem:harmonic-map-flow-ricci-deturck}
\label{dtuniqtouniq}

If two Ricci flows agree at $t_0$ and their associated Ricci--DeTurck
flows agree on $[t_0,T]$, then the Ricci flows agree on $[t_0,T]$.
\end{corollary}
\begin{proof}
\sketch Equality of pullback metrics identifies the two harmonic-map-flow
diffeomorphisms by uniqueness, and hence the original metrics.
\end{proof}

\section{Harmonic map flow}
Let $g(t)=dr^2+f(r,t)^2d\sigma$,
\begin{equation}\label{eq2.4}g(t)=dr^2+f(r,t)^2d\sigma.\end{equation}

\begin{claim}[The radial coordinate $r$ as a function of $\hat r$ and $t$]
\label{claim:radial-coordinate-properties}
\uses{def:standard-ricci-flow}
\label{cl12.20}

$r=r(\hat r,t)$ is smooth away from $\hat r=0$, odd in $\hat r$, with
$r(0,t)=0$, $\partial_{\hat r}r(0,t)>0$, and
$r^2=\hat r^2\xi(\hat r^2,t)$ for a positive smooth $\xi$.
\end{claim}

\begin{equation}\label{ansatz}
\psi_t(r,\theta)=(\rho(r,t),\theta),\qquad\psi_{t_0}=\Id.
\end{equation}

\begin{lemma}[Even and odd radial functions extend smoothly]
\label{lem:even-odd-radial-functions}
If $a(w,t)$ is smooth for $w\ge0$, then $a(r^2,t)$ is smooth and even
in $r$; $r a(r^2,t)$ is smooth and odd. Conversely, smooth radial even
and odd functions have these forms.
\end{lemma}

\begin{corollary}[$r^2$ is a smooth function of the ambient coordinates]
\label{cor:r-squared-smooth-function}
\uses{claim:radial-coordinate-properties, lem:even-odd-radial-functions}
\label{rcor}

$r^2(\hat r,t)$ is smooth on $\Ar^n$, and $\hat r$ is smooth in $(r,t)$
for $r>0$.
\end{corollary}
Define
\begin{equation}\label{Bdefn}B(w,t)=\frac12\int_0^w\xi(u,t)\,du.\end{equation}

\begin{claim}[$f(r,t)$ is a smooth odd function]
\label{claim:f-odd-smooth-function}
\uses{cor:r-squared-smooth-function}

$f$ extends smoothly and oddly in $r$.
\end{claim}
\begin{corollary}[Factorization $f(r,t)=r\,h(r^2,t)$]
\label{cor:f-factorization-rh}
\uses{claim:f-odd-smooth-function}

There is smooth $h(w,t)$ with $f(r,t)=rh(r^2,t)$.
\end{corollary}

The radial harmonic map equation is
\begin{equation}\label{eqrothar}
\partial_t\rho=f^{1-n}\partial_r(f^{n-1}\partial_r\rho)-(n-1)f^{-2}f(\rho,t_0).
\end{equation}
Writing $\rho=re^{\tilde\rho}$ yields
\begin{equation}\label{newform}
\partial_t\tilde\rho=-\partial_r^2\tilde\rho-\frac{n+1}{r}\partial_r\tilde\rho+\cdots,
\end{equation}
where the omitted terms are collected in the smooth function
\begin{equation}\label{defofG}G(\tilde\rho,w,t).\end{equation}
Equivalently on $\Ar^{n+2}$,
\begin{equation}\label{betterharMapeq}
\partial_t\tilde\rho=\Delta\tilde\rho+\nabla((n-1)\tilde h-B)\cdot\nabla\tilde\rho+|\nabla\tilde\rho|^2+G(\tilde\rho,x,t).
\end{equation}

\begin{remark}[Why we work on $\mathbb{R}^{n+2}$]
\label{rem:choice-of-dimension-n-plus-2}
This dimension supplies the radial term $(n+1)r^{-1}\partial_r$ and
makes all coefficients of \eqref{betterharMapeq} smooth at the origin.
\end{remark}

\begin{claim}[Asymptotics of the harmonic map flow coefficients]
\label{claim:harmonic-map-flow-asymptotics}
\uses{prop:standard-flow-asymptotic-cylinder}
\label{asympt}

At infinity, $e^{\tilde h(r^2,t)}\sim((1-t)r)^{-1}$,
$\tilde h\sim-\log r$, $\partial_w\tilde h\sim-(2r^2)^{-1}$,
$r^{-1}\partial_tr\sim C/r$, $\partial_rB\sim C$, and
$|G(\tilde\rho,r^2,t)|\le C_*$ with $C_*$ depending only on
$\sup\{|\tilde\rho|,|\tilde h|\}$.
\end{claim}
\begin{proof}
\uses{prop:standard-flow-asymptotic-cylinder}
\sketch The cylindrical asymptotics give the first four estimates; the
definition of $B$ gives the fifth, and substitution into $G$ gives the last.
\end{proof}

\begin{proposition}[Short-time existence for the harmonic map flow]
\label{prop:harmonic-map-flow-short-time-existence}
\uses{def:standard-ricci-flow}

For every $t_0<T_1$ for which a standard solution is defined on $[0,T_1]$,
\eqref{betterharMapeq} with $\tilde\rho(\cdot,t_0)=0$ has a solution on
$[t_0,T]$ for some $T>t_0$.
\end{proposition}
\begin{proof}
\sketch Shift $t_0$ to $0$ and iterate the heat-kernel formula
\begin{equation}\label{rhoIntg}
 \tilde\rho_i=\int_0^t\!\int_{\Ar^{n+2}}H(x,y,t-s)F(y,\tilde\rho_{i-1},\nabla\tilde\rho_{i-1},s)\,dy\,ds,
\end{equation}
which satisfies
\begin{equation}\label{rhodiffeq}\partial_t\tilde\rho_i=\Delta\tilde\rho_i+F_{i-1},\qquad\tilde\rho_i(\cdot,0)=0.\end{equation}
For short time,
\begin{equation}\label{rho_i est}|\tilde\rho_i|\le C_1,\qquad|\nabla\tilde\rho_i|\le C_2.\end{equation}
Differences satisfy \eqref{rhoiminus} and the Lipschitz estimate \eqref{Flip},
so they contract in $C^1$; the limit satisfies \eqref{rhoInfIntg} and
the Schauder estimate \eqref{rhoHolder}.
\end{proof}
\begin{equation}\label{rhoiminus}
\partial_t(\tilde\rho_i-\tilde\rho_{i-1})=\Delta(\tilde\rho_i-\tilde\rho_{i-1})+F_{i-1}-F_{i-2}.
\end{equation}
\begin{equation}\label{Flip}|F_{i-1}-F_{i-2}|\le C_4|\tilde\rho_{i-1}-\tilde\rho_{i-2}|+C_5|\nabla\tilde\rho_{i-1}-\nabla\tilde\rho_{i-2}|.\end{equation}
\begin{equation}\label{rhoInfIntg}\tilde\rho_\infty=\int_0^t\!\int H(x,y,t-s)F(y,\tilde\rho_\infty,\nabla\tilde\rho_\infty,s)\,dy\,ds.\end{equation}
\begin{equation}\label{rhoHolder}|\nabla\tilde\rho_\infty(x,t)-\nabla\tilde\rho_\infty(y,s)|\le C_7(|x-y|^\alpha+|t-s|^{\alpha/2}).\end{equation}

\subsection{Decay}\label{harmsec}
The iteration also gives
\begin{equation}\label{rho_iDcay}|\tilde\rho_i|+|\nabla\tilde\rho_i|\le\frac\lambda{(1+|x|)^2},\end{equation}
and hence
\begin{equation}\label{rhodecay}|\tilde\rho_\infty|+|\nabla\tilde\rho_\infty|\le\frac\lambda{(1+|x|)^2}.\end{equation}

\begin{proposition}[Harmonic map flow with quadratic decay at infinity]
\label{prop:harmonic-map-flow-existence-decay}
\uses{prop:harmonic-map-flow-short-time-existence, claim:harmonic-map-flow-asymptotics}
\label{harmapflow}

For every standard solution and $t_0<T$, there is a rotationally symmetric
harmonic map flow $\psi_t(x)=xe^{\tilde\rho(x,t)}$ on $[t_0,T']$ with
$\psi_{t_0}=\Id$ and $|\nabla^i\tilde\rho|(x,t)\le C_{14}(1+|x|)^{-2}$
for $0\le i\le3$.
\end{proposition}

\begin{proposition}[Uniqueness for the Ricci-DeTurck flow on noncompact manifolds]
\label{prop:ricci-deturck-uniqueness-noncompact}
\uses{def:standard-ricci-flow}
\label{RDuniq}

Let bounded Ricci--DeTurck flows $\hat g_1,\hat g_2$ on complete noncompact
$M$ have common initial metric $g$, uniform $C^2(g)$ and bilipschitz bounds,
and the same sequential $C^1$ asymptotics along an exhaustion. Then
$\hat g_1=\hat g_2$.
\end{proposition}
\begin{proof}
\sketch Their difference satisfies
\begin{equation}\label{eqestforunique}
 \partial_t|\hat g_1-\hat g_2|_g^2\le\hat g_1^{\alpha\beta}\tilde\nabla_\alpha\tilde\nabla_\beta|\hat g_1-\hat g_2|_g^2+C_{15}|\hat g_1-\hat g_2|_g^2.
\end{equation}
On exhaustion boundaries the difference obeys
\begin{equation}\label{eq est for uniq
bddy}
\sup_{\partial\Omega_b}|\hat g_1(t')-\hat g_2(t')|_g^2\le\epsilon.
\end{equation}
The maximum principle, then $b\to\infty$ and $\epsilon\to0$, proves equality.
\end{proof}

\begin{corollary}[Agreement of Ricci-DeTurck flows for standard solutions]
\label{cor:standard-solution-deturck-agreement}
\uses{prop:harmonic-map-flow-existence-decay, prop:ricci-deturck-uniqueness-noncompact}
\label{duniq}

If standard solutions $g_1,g_2$ agree on $[0,t_0]$, their associated
Ricci--DeTurck flows agree on $[t_0,T']$ for some $T'>t_0$.
\end{corollary}

\section{Completion}
For two standard flows, extend equality past a closed maximal agreement
interval using Corollary~\ref{cor:standard-solution-deturck-agreement} and
Corollary~\ref{cor:ricci-deturck-uniqueness-implies-ricci-uniqueness}. If one
maximal interval ends earlier, bounded curvature of the other gives an
extension. Thus the flows agree throughout their common maximal interval.

\begin{theorem}[Canonical neighborhoods in the standard solution]
\label{thm:standard-solution-canonical-neighborhoods}
\uses{thm:standard-solution-existence-uniqueness, prop:standard-flow-kappa-noncollapsed}
\label{R_0cannbhd}

For $0<\epsilon<1$ there is $r>0$ such that $R(x_0,t_0)\ge r^{-2}$
implies $t_0>r^2$ and a strong $(C(\epsilon),\epsilon)$-canonical
neighborhood at $(x_0,t_0)$. A strong neck there extends backward for
rescaled time $1+\epsilon$.
\end{theorem}
\begin{proof}
\uses{prop:standard-flow-kappa-noncollapsed,kaplimit,smlmtflow,limitcannbhd}
\sketch A highest-curvature counterexample sequence has $Q_n\to\infty$.
Noncollapsing, positive pinching, and the inductive canonical-neighborhood
bound meet the compactness hypotheses; its $\kappa$-solution limit transfers
the required neighborhood back, a contradiction.
\end{proof}

\begin{claim}[Volume ratio of the rescaled limit is small]
\label{claim:kappa-solution-volume-ratio-small}
\uses{asympvol}

For each $\epsilon>0$ some $A<\infty$ satisfies
$\Vol(B_{Q_ng}(x,t_n,A))<\epsilon A^3$ for all large $n$.
\end{claim}

\begin{theorem}[The standard flow exists exactly on $[0,1)$]
\label{thm:standard-solution-time-interval-one}
\uses{thm:standard-solution-canonical-neighborhoods, prop:standard-flow-asymptotic-cylinder}

$T=1$.
\end{theorem}
\begin{proof}
\uses{cor:standard-flow-time-bound-one,thm:standard-solution-canonical-neighborhoods,prop:standard-flow-asymptotic-cylinder,shiw/deriv,kaplimit,asympvol,thm:bishop-gromov,lem:distance-nonincreasing}
\sketch Assume $T<1$. Cylindrical asymptotics and canonical neighborhoods
give a smooth limit outside a compact set. A blowup point has a
$\kappa$-solution limit, so the preceding claim and Bishop--Gromov force the
volume of a fixed positive-volume subset
of the exterior limit to vanish, a contradiction. Thus $T\ge1$; combine
with $T\le1$.
\end{proof}

\section{Corollaries}
\begin{proposition}[Scalar curvature lower bound $R\ge c/(1-t)$]
\label{prop:standard-solution-scalar-curvature-lower-bound}
\uses{thm:standard-solution-canonical-neighborhoods}
\label{Restim}

There is $c>0$ such that $R(p,t)\ge c/(1-t)$ everywhere.
\end{proposition}
\begin{proof}
\uses{thm:standard-solution-canonical-neighborhoods,lem:distance-nonincreasing,controlnbhd,canonvary,narrows,omega,kaplimit,asympvol,thm:bishop-gromov}
\sketch A global metric limit at time $1$ would produce arbitrarily small
canonical necks, contradicting the narrow-neck estimate. A point with bounded
curvature would likewise contradict the zero asymptotic volume of a blowup
$\kappa$-solution and Bishop--Gromov. Canonical-neighborhood derivative
control then integrates to the stated uniform lower bound.
\end{proof}

\begin{theorem}[Trichotomy of canonical neighborhoods in the standard solution]
\label{thm:standard-solution-trichotomy}
\uses{thm:standard-solution-canonical-neighborhoods, prop:standard-solution-scalar-curvature-lower-bound}
\label{stdsolncannbhd}

For every $\epsilon>0$ there is $C'(\epsilon)<\infty$ such that every
$(x,t)$ lies either in the core of a $(C',\epsilon)$-cap, at the center
of an evolving $\epsilon$-neck starting at $0$ and initially disjoint from
the surgery cap, or at the center of an evolving $\epsilon$-neck defined
for rescaled time $1+\epsilon$.
\end{theorem}
\begin{proof}
\uses{thm:standard-solution-canonical-neighborhoods,prop:standard-solution-scalar-curvature-lower-bound,prop:standard-flow-asymptotic-cylinder}
\sketch Apply the high-curvature theorem above. At bounded curvature, the
scalar lower bound restricts time away from $1$; cylindrical asymptotics give
the outer necks and compactness gives caps on the remaining compact set.
\end{proof}

\begin{remark}[Evolving necks of rescaled time greater than one]
\label{rem:evolving-neck-rescaled-time}
\uses{def:evolving-epsilon-neck}

This is compatible with the time interval $[0,1)$ because the neck scale is
$R(x,t)^{-1/2}$, which tends to $0$ as $t\to1$.
\end{remark}

\begin{corollary}[Points near the tip lie in a canonical cap]
\label{cor:standard-solution-cap-near-tip}
\uses{thm:standard-solution-trichotomy}
\label{stdsolncap}

If $t\le R(q,t)^{-1}(1+\epsilon)$ and
$(q,0)\in B(p_0,0,\epsilon^{-1}/2+A_0+5)$, then $(q,t)$ lies in a
$(C'(\epsilon),\epsilon)$-cap.
\end{corollary}

\begin{remark}[The tip $p_0$ and the constant $A_0$]
\label{rem:tip-and-a0-definitions}
\uses{lem:standard-initial-metric-asymptotic-bounds}

$p_0$ is the origin (the surgery-cap tip), and $A_0$ is from

\end{remark}

\begin{corollary}[Trichotomy for flows converging to the standard solution]
\label{cor:generalized-flow-limit-trichotomy}
\uses{thm:standard-solution-trichotomy}
\label{stdsolnlimit}

Let generalized flows $({\mathcal M}_n,G_n)$ converge geometrically on
$U_n\times[0,\theta)$, $0<\theta<1$, to the standard solution. For large
$n$, every point of the image lies in a $(C'(\epsilon),\epsilon)$-cap,
is the center of a strong $\epsilon$-neck, or is the center of an evolving
$\epsilon$-neck with initial time $0$.
\end{corollary}
\begin{proof}
\uses{thm:standard-solution-trichotomy,canonvary}
\sketch Apply the standard-flow trichotomy and stability of canonical
neighborhoods under geometric convergence.
\end{proof}

\begin{lemma}[Distance-decreasing property of the standard initial metric]
\label{lem:standard-initial-metric-distance-decreasing}
\uses{def:standard-initial-metric}
\label{decrease}

Let $h_0$ be twice the usual metric on the unit sphere in $T_0\Ar^3$ and
let $\rho(x,s)$ be radial exponential coordinates for $g_0$. Then
$\rho^*g_0\le h_0\times ds^2$.
\end{lemma}
\begin{proof}
\sketch The radial component is $ds^2$ and each rotational cross-section
has radius at most $\sqrt2$, which gives the tangential inequality.
\end{proof}
