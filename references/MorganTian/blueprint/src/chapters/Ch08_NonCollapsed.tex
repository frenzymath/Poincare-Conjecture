\section{A non-collapsing result for generalized Ricci flows}

\begin{theorem}[$\kappa$-non-collapsing for generalized Ricci flows]
\label{thm:kappa-noncollapsed-generalized-flow}
\uses{def:generalized-ricci-flow, def:compatible-embedding}
\label{THM}

Fix $\bar\tau_0,l_0,V>0$. There is
$\kappa=\kappa(\bar\tau_0,V,l_0,n)>0$ with the following property.
Let $({\mathcal M},G)$ be a generalized $n$-dimensional Ricci flow, fix
$x\in{\mathcal M}$ with ${\bf t}(x)=T$, and suppose
$0<\tau_0\leq\bar\tau_0$ and $0<r\leq\sqrt{\tau_0}$. Assume:
\begin{enumerate}
\item[(1)] $B(x,T,r)\subset M_T$ has compact closure;
\item[(2)] there is an embedding
$B(x,T,r)\times[T-r^2,T]\subset{\mathcal M}$ compatible with time and the
time vector field;
\item[(3)] $|\Rm|\leq r^{-2}$ on this parabolic neighborhood;
\item[(4)] an open set
$\widetilde W\subset\widetilde{\mathcal U}_x(\tau_0)\subset T_xM_T$ has the
property that every associated $\mathcal L$-geodesic on $[0,\tau_0]$ has
reduced length at most $l_0$;
\item[(5)] for
$W(\tau)={\mathcal L}\operatorname{exp}_x^\tau(\widetilde W)$,
$\Vol W(\tau_0)\geq V$.
\end{enumerate}
Then
\[
  \Vol B(x,T,r)\geq\kappa r^n.
\]
\end{theorem}

\begin{proof}
  \uses{def:reduced-volume, thm:reduced-volume-monotone, thm:kappa-noncollapsed-generalized-flow, prop:reduced-volume-upper-bound, lem:reduced-volume-small-directions, claim:l-geodesic-stays-in-ball, claim:volume-comparison-time-slices, thm:shi-first-derivative}
\sketch

Reduced-volume monotonicity and hypotheses (4)--(5) give, for
$0<\tau\leq\tau_0$,
\begin{equation}
\label{redvoleqn}
  \widetilde V_x(W(\tau))
  \geq\bar\tau_0^{-n/2}Ve^{-l_0}.
\end{equation}
Write
$\Vol B(x,T,r)=\varepsilon^nr^n$. If
$\varepsilon\leq\varepsilon_0$, Proposition~\ref{step}, applied at
$\tau_1=\varepsilon r^2$, gives
\[
  3\varepsilon^{n/2}>
  \widetilde V_x(W(\tau_1))
  \geq\bar\tau_0^{-n/2}Ve^{-l_0}.
\]
Thus either $\varepsilon>\varepsilon_0$ or
\[
  \varepsilon\geq
  \frac{V^{2/n}e^{-2l_0/n}}{3^{2/n}\bar\tau_0}.
\]
Taking the $n$th power of the smaller lower bound gives the required
$\kappa$.
\end{proof}

\begin{proposition}[Upper bound for the reduced volume of $W(\tau_1)$]
\label{prop:reduced-volume-upper-bound}
\uses{thm:kappa-noncollapsed-generalized-flow}
\label{step}

There is
$0<\varepsilon_0\leq1/(4n(n-1))$, depending on
$\bar\tau_0$ and $l_0$, such that if
\[
  \varepsilon=rac{\Vol(B(x,T,r))^{1/n}}r\leq\varepsilon_0,
  \qquad \tau_1=\varepsilon r^2,
\]
then
\[
  \widetilde V_x(W(\tau_1))<3\varepsilon^{n/2}.
\]
\end{proposition}

\begin{proof}
\sketch
Split
\[
  \widetilde W_{\mathrm{sm}}
  =\left\{Z\in\widetilde W:|Z|\leq
  \frac1{8\sqrt\varepsilon}\right\},
  \qquad
  \widetilde W_{\mathrm{lg}}=\widetilde W\setminus\widetilde W_{\mathrm{sm}},
\]
and let $W_{\mathrm{sm}}(\tau_1)$ and $W_{\mathrm{lg}}(\tau_1)$ be their
$\mathcal L$-exponential images. Lemmas~\ref{small} and~\ref{large} give
\[
  \widetilde V_x(W_{\mathrm{sm}}(\tau_1))\leq2\varepsilon^{n/2},
  \qquad
  \widetilde V_x(W_{\mathrm{lg}}(\tau_1))\leq\varepsilon^{n/2}.
\]
The two images are disjoint and cover $W(\tau_1)$ up to a null set, which
proves the estimate.
\end{proof}

\subsection{Small initial vectors}

\begin{lemma}[Reduced volume bound for the small-vector region]
\label{lem:reduced-volume-small-directions}
\uses{prop:reduced-volume-upper-bound}
\label{small}

For $\tau_1=\varepsilon r^2$, there is
$\varepsilon_0=\varepsilon_0(\bar\tau_0)>0$ such that
$\varepsilon\leq\varepsilon_0$ implies
\[
  \int_{\widetilde W_{\mathrm{sm}}}
  \tau_1^{-n/2}e^{-\widetilde l(Z,\tau_1)}
  {\mathcal J}(Z,\tau_1)\,dZ
  \leq2\varepsilon^{n/2}.
\]
\end{lemma}

\begin{proof}
\sketch
\uses{claim:l-geodesic-stays-in-ball,claim:volume-comparison-time-slices}
By Claim~\ref{inside}, the relevant $\mathcal L$-geodesics remain in
$B(x,T,r/2)$ until $\tau_1$. Since $|\Rm|\leq r^{-2}$ there,
$R\geq-n(n-1)r^{-2}$, and hence
$l_x(q,\tau_1)\geq-n(n-1)\varepsilon/3$. Therefore
\begin{equation}
\label{volineq}
  \widetilde V_x(W_{\mathrm{sm}}(\tau_1))
  \leq\varepsilon^{-n/2}r^{-n}e^{n(n-1)\varepsilon/3}
  \Vol_{g(T-\tau_1)}B(x,T,r).
\end{equation}
Claim~\ref{volvomp} and
$\Vol_{g(T)}B(x,T,r)=\varepsilon^nr^n$ bound the right-hand side by
$1.1e^{1/12}\varepsilon^{n/2}<2\varepsilon^{n/2}$ after decreasing
$\varepsilon_0$.
\end{proof}

\begin{corollary}[Reduced volume of $W_{\mathrm{sm}}(\tau_1)$ is small]
\label{cor:reduced-volume-small-bound}
\uses{lem:reduced-volume-small-directions}

For sufficiently small
$\varepsilon_0=\varepsilon_0(\bar\tau_0)>0$, if
$\varepsilon\leq\varepsilon_0$, then
\[
  \widetilde V_x(W_{\mathrm{sm}}(\tau_1))\leq2\varepsilon^{n/2}.
\]
\end{corollary}

\begin{claim}[Gradient of scalar curvature and metric comparison bounds]
\label{claim:scalar-curvature-gradient-metric-bound}
\uses{thm:kappa-noncollapsed-generalized-flow}
\label{nabR}

There are a universal $\varepsilon'_0>0$ and a constant
$C_1=C_1(n)<\infty$ such that, if
$\varepsilon\leq\varepsilon'_0$, then for
$y\in B(x,T,r/2)$ and $t\in[T-\tau_1,T]$,
\[
  |\nabla R(y,t)|\leq\frac{C_1}{r^3},
  \qquad
  (1-C_1\varepsilon)g(y,T)
  \leq g(y,t)\leq
  (1+C_1\varepsilon)g(y,T).
\]
\end{claim}

\begin{proof}
\sketch
\uses{thm:shi-first-derivative}
After parabolic rescaling by $r^{-2}$, Shi's first-derivative estimate on the
unit parabolic ball gives the gradient bound; rescaling back contributes
$r^{-3}$. The curvature hypothesis also gives
$|\Ric|\leq(n-1)r^{-2}$. Integrating
$\partial_tg=-2\Ric$ over an interval of length at most
$\tau_1=\varepsilon r^2$ gives exponential bounds
$e^{-2(n-1)\varepsilon}g(T)\leq g(t)\leq
e^{2(n-1)\varepsilon}g(T)$, which imply the displayed comparison for small
$\varepsilon$.
\end{proof}

\begin{claim}[$\mathcal L$-geodesic norm comparison estimate]
\label{claim:l-geodesic-norm-comparison}
\uses{claim:scalar-curvature-gradient-metric-bound}

There is
$0<\varepsilon_0\leq1/(4n(n-1))$, depending on $\bar\tau_0$, such that the
following holds. If $\varepsilon\leq\varepsilon_0$,
$\tau'_1\leq\tau_1$, and the $\mathcal L$-geodesic $\gamma_Z$ remains in
$B(x,T,r/2)$ for $\tau<\tau'_1$, then
\[
  \left||\sqrt\tau X(\tau)|_{g(T)}-|Z|\right|
  \leq2\varepsilon(1+|Z|)
  \qquad(0<\tau<\tau'_1),
\]
where $X=\gamma_Z'$.
\end{claim}

\begin{proof}
\sketch
\uses{claim:scalar-curvature-gradient-metric-bound,lem:l-geodesic-min-max-estimate}
Apply the $\mathcal L$-geodesic min--max estimate using
$|\nabla R|\leq C_1r^{-3}$, $|\Ric|\leq(n-1)r^{-2}$, and time length
$\varepsilon r^2$. Its exponential errors are
$e^{2C_1\varepsilon^2}-1=O(\varepsilon^2)$, giving
\[
  \left||\sqrt\tau X(\tau)|_{g(T-\tau)}-|Z|\right|
  \leq\varepsilon(1+|Z|)
\]
after decreasing $\varepsilon_0$. The metric comparison in Claim~\ref{nabR}
gives the asserted $g(T)$ estimate.
\end{proof}

\begin{claim}[$\mathcal L$-geodesics with small initial vector stay in the half-ball]
\label{claim:l-geodesic-stays-in-ball}
\uses{claim:l-geodesic-norm-comparison}
\label{inside}

Let $\varepsilon_0$ be as in the preceding claim. If
$\varepsilon\leq\varepsilon_0$, $\tau_1=\varepsilon r^2$, and
$|Z|\leq(8\sqrt\varepsilon)^{-1}$, then
$\gamma_Z(\tau)\in B(x,T,r/2)$ for every $0\leq\tau\leq\tau_1$.
\end{claim}

\begin{proof}
\sketch
\uses{claim:l-geodesic-norm-comparison}
While the curve remains in the half-ball, the preceding estimate and
$\varepsilon\leq1/8$ give
$|\sqrt\tau X(\tau)|_{g(T)}\leq(4\sqrt\varepsilon)^{-1}$. If $\tau'<\tau_1$
were the first exit time, then
\[
  d_{g(T)}(x,\gamma_Z(\tau'))
  \leq\frac1{4\sqrt\varepsilon}
  \int_0^{\tau'}\tau^{-1/2}\,d\tau
  <\frac r2,
\]
contradicting the definition of $\tau'$.
\end{proof}

\begin{claim}[Volume comparison between nearby time-slices]
\label{claim:volume-comparison-time-slices}
\uses{thm:kappa-noncollapsed-generalized-flow}
\label{volvomp}

There is a universal $\varepsilon_0>0$ such that if
$\varepsilon\leq\varepsilon_0$, then for every open
$U\subset B(x,T,r)$ and $0\leq\tau_1\leq\tau_0$,
\[
  0.9\leq
  \frac{\Vol_{g(T)}U}{\Vol_{g(T-\tau_1)}U}
  \leq1.1.
\]
\end{claim}

\begin{proof}
\sketch
\uses{claim:scalar-curvature-gradient-metric-bound}
The metric comparison in Claim~\ref{nabR} gives the corresponding determinant
and volume-form bounds; decrease $\varepsilon_0$ until these lie in
$[0.9,1.1]$.
\end{proof}

\subsection{Large initial vectors}

\begin{lemma}[Reduced volume bound for the large-vector region]
\label{lem:reduced-volume-large-directions}
\uses{prop:reduced-volume-upper-bound}
\label{large}

There is a universal $\varepsilon_0>0$ such that
$\varepsilon\leq\varepsilon_0$ implies
\[
\begin{split}
  \widetilde V_x(W_{\mathrm{lg}}(\tau_1))
  &\leq
  \int_{\widetilde U(\tau_1)\cap
  \{|Z|\geq(8\sqrt\varepsilon)^{-1}\}}
  \tau_1^{-n/2}e^{-\widetilde l(Z,\tau_1)}
  {\mathcal J}(Z,\tau_1)\,dZ\\
  &\leq\varepsilon^{n/2}.
\end{split}
\]
\end{lemma}

\begin{proof}
\sketch
\uses{prop:reduced-volume-integrand-monotone}
Monotonicity of the reduced-volume density bounds its pullback by
$2^ne^{-|Z|^2}$. Hence the integral is at most the Gaussian tail outside the
ball of radius $(8\sqrt\varepsilon)^{-1}$. Comparing that ball with the
centered cube of half-side $(8\sqrt{n\varepsilon})^{-1}$ gives
\[
  \widetilde V_x(W_{\mathrm{lg}}(\tau_1))
  \leq(4\pi)^{n/2}\frac n2
  e^{-1/(64n\varepsilon)}.
\]
For sufficiently small universal $\varepsilon_0$, this is at most
$\varepsilon^{n/2}$.
\end{proof}

\section{Application to compact Ricci flows}

\begin{theorem}[$\kappa$-non-collapsing for compact 3-dimensional Ricci flows]
\label{thm:kappa-noncollapsed-compact-3-manifolds}
\uses{def:ricci-flow}

Fix $\omega,T_0>0$. There is $\kappa=\kappa(\omega,T_0)>0$ such that
the following holds. Let $(M,g(t))$, $0\leq t<T\leq T_0$, be a Ricci flow
on a compact $3$-manifold satisfying
\[
  |\Rm(p,0)|\leq1,
  \qquad
  \Vol B(p,0,1)\geq\omega
  \quad\text{for every }p\in M.
\]
If $t_0\leq T$, $r^2\leq t_0$, and
\[
  |\Rm(q,t)|\leq r^{-2}
  \quad\text{on}\quad
  B(p,t_0,r)\times[t_0-r^2,t_0],
\]
then
$\Vol B(p,t_0,r)\geq\kappa r^3$.
\end{theorem}

\begin{proof}
\sketch
\uses{thm:bishop-gromov,thm:kappa-noncollapsed-generalized-flow,prop:normalized-flow-short-time-bounds,thm:min-l-bound-n-half,prop:l-exp-lipschitz-complete}
Parabolic rescaling and Bishop--Gromov reduce to
$t_0\geq1$ and $r\leq\sqrt{t_0}/2$, changing only the final universal
constant. Normalized short-time estimates give universal curvature and volume
bounds on $M\times[0,2^{-4}]$, hence uniform equivalence of the metrics there.
Choose an $\mathcal L$-geodesic from $(p,t_0)$ to
$(q_0,2^{-4})$ with reduced length at most $3/2$. Concatenating it with
short $g(0)$-geodesics from $q_0$ to points of $B(q_0,0,1)$ gives a uniform
reduced-length bound at backward time
$\tau_0=t_0-2^{-5}$. This ball has a uniform positive volume, and the regular
domain of the $\mathcal L$-exponential has full measure. Since
$r^2\leq t_0/4<\tau_0$, Theorem~\ref{THM} gives the claimed lower bound.
\end{proof}
