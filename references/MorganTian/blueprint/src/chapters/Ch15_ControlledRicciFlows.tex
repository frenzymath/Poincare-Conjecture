\chapter{Controlled Ricci flows with surgery}
\label{chap:controlled-ricci-flows}

\subsection{Normalized initial conditions}

A compact connected Riemannian $3$-manifold $(M,g(0))$ is
\emph{normalized} if
\[
  |\Rm(x,0)|\leq1,
  \qquad
  \Vol B(x,0,1)\geq\omega/2
  \quad(x\in M),
\]
where $\omega$ is the Euclidean unit-ball volume. Every compact metric becomes
normalized after multiplication by a sufficiently large constant.

\begin{claim}[Normalized initial data give short-time existence with curvature and volume control]
\label{claim:normalized-initial-existence}
\uses{thm:local-existence-uniqueness, prop:smooth-patching, thm:shi-derivative-estimates, prop:normalized-flow-short-time-bounds}
\label{newkappa0r0t0}

There is $\kappa_0>0$ such that a normalized initial metric has a Ricci flow
on $[0,2^{-4}]$ with
\[
  |R(x,t)|\leq2,
  \qquad
  \Vol B(x,t,r)\geq\kappa_0r^3
\]
for all $x$, $t\in[0,2^{-4}]$, and $r\leq\epsilon$.
\end{claim}

\section{Gluing evolving necks}

\begin{proposition}[Gluing an evolving neck to a strong neck produces a longer strong neck]
\label{prop:neck-gluing}
\uses{def:strong-canonical-neighborhood-assumption, def:evolving-epsilon-neck}

\label{neckglue}
There is $0<\beta<1/2$ such that, for $\epsilon<1$, an evolving
$\beta\epsilon$-neck on $[-t_0,0]$ centered at $x$ with $R(x,0)=1$ and a
strong $\beta\epsilon/2$-neck on $(-t_1,-t_0]$ centered at the identified point
at time $-t_0$ glue to contain a strong $\epsilon$-neck centered at $(x,0)$.
\end{proposition}

\begin{proof}
\sketch
If the conclusion failed, take $\beta_n\to0$ and counterexamples. Smooth
compactness gives an evolving round cylinder. The strong-neck part extends the
limit for at least one unit of backward time; whether the limiting junction time
is positive or zero, compact convergence on a fixed cylinder forces the union to
contain a strong $\epsilon$-neck, a contradiction.
\end{proof}

\subsection{First assumptions}
\label{firstassumpt}

Fix $\epsilon>0$ sufficiently small, with
\[
  \epsilon\leq\min\left(\frac1{200},\frac1{\sqrt D(A_0+5)},
  \frac{\bar\epsilon_1}2,\frac{\bar\epsilon'}2,\epsilon_0\right),
\]
and let $\beta$ be as in Proposition~\ref{neckglue}. Choose
$C$ larger than the canonical-neighborhood constants supplied by the
$\kappa$-solution and standard-solution analyses.

We impose the following assumptions on a $3$-dimensional Ricci flow with
surgery $({\mathcal M},G)$:
\begin{enumerate}
\item[(1)] every time-slice is compact and contains no embedded $\Ar P^2$ with
trivial normal bundle;
\item[(2)] singular times form a discrete subset of $\Ar$;
\item[(3)] $0$ is the initial time and $(M_0,G(0))$ is normalized;
\item[(4)] at every singular time $t$, the exposed region $E_t$ is a finite
disjoint union of $3$-balls, so its boundary $\Sigma_t$ is a finite union of
$2$-spheres;
\item[(5)] every component of $\Sigma_t$ is the central sphere of a strong
$\delta(t)$-neck in the maximal pre-surgery extension, where $\delta(t)>0$ is
nonincreasing;
\item[(6)] for $t^-<t$ sufficiently close to each singular time, every point
of the disappearing region $D_{t^-}$ has a strong $(C,\epsilon)$-canonical
neighborhood;
\item[(7)] between successive singular times, the flow is a maximal ordinary
Ricci flow.
\end{enumerate}
At a singular time, $M_t=C_t\cup_{\Sigma_t}E_t$, where $C_t$ is the continuing
region: the maximal subset admitting a compatible embedding
$C_t\times(t-\delta,t]$.

\section{Topological consequences of Assumptions (1)--(7)}

\begin{proposition}[Topological effect of a surgery time on the earlier time-slice]
\label{prop:surgery-topological-type}
\label{surgerytoptype}
Suppose $({\mathcal M},G)$ satisfies Assumptions (1)--(7), and let $t$ be a
singular time. For $t^-<t$ sufficiently close, $M_{t^-}$ is obtained from the
disjoint union of $M_t$, finitely many $2$-sphere bundles over $S^1$, and finitely
many closed $3$-manifolds of constant positive curvature by connected-sum
operations among components.
\end{proposition}

\begin{proof}
\sketch
\uses{prop:classification-covered-by-necks-caps,prop:balanced-chain-covers-connected-set}
The exposed $3$-balls show that passing from $M_{t^-}$ to $M_t$ removes a
disappearing region and caps each boundary sphere. Components of the disappearing
region that are whole components are spherical space-forms,
$\Ar P^3\#\Ar P^3$, or $S^2$-bundles over $S^1$. A noncompact component covered
by canonical necks and caps is an $\epsilon$-tube, a capped tube, a $3$-ball,
or $\Ar P^3\setminus B^3$. Replacing these by balls respectively performs no
change, a connected-sum surgery (possibly deleting an $S^2\times S^1$ factor),
or a connected-sum surgery deleting an $\Ar P^3$ factor. This gives the stated
decomposition.
\end{proof}

\begin{corollary}[Geometrization is inherited across surgery times]
\label{cor:geometrization-inherited-across-surgery}
\uses{prop:surgery-topological-type}

\label{Tgood0good}
If a time-slice $M_T$ of a flow satisfying Assumptions (1)--(7) satisfies
Thurston's Geometrization Conjecture, then every earlier slice and the initial
manifold do. If $M_T=\emptyset$, the initial manifold is a connected sum of
$2$-sphere bundles over $S^1$ and $3$-dimensional space-forms. In particular,
an initial connected simply connected manifold is $S^3$, and an initial
manifold with finite fundamental group is a space-form.
\end{corollary}

\begin{proof}
\sketch
Induct backward over the finitely many surgery times using
Proposition~\ref{surgerytoptype}. Space-forms already satisfy geometrization;
the two $S^2$-bundles over $S^1$ are obtained by gluing the ends of
$S^2\times I$ by the identity or antipodal map and have the homogeneous
$S^2\times\Ar$ geometry. The extinction conclusions follow from the same
backward induction and the fundamental-group hypotheses.
\end{proof}

\section{Further conditions on surgery}

\begin{definition}[Canonical neighborhood and surgery control parameters; surgery scale function]
\label{def:surgery-scale-function}
\uses{thm:local-surgery, thm:deep-delta-necks-in-horn}

\label{hrestricted}
Let $r(t)>0$ be the canonical-neighborhood parameter and
$\overline\delta(t)>0$ the nonincreasing surgery-control parameter. Set
\[
  \rho(t)=\overline\delta(t)r(t),
  \qquad
  h(t)=h(\rho(t),\overline\delta(t))
  \leq\overline\delta(t)^2r(t),
\]
where $h$ is supplied by Theorem~\ref{hexist}, and require
$h(0)\leq R_0^{-1/2}$ from Theorem~\ref{LOCALSURGERY}. Let
$\kappa(t)>0$ be the noncollapsing parameter. The three functions
$r,\overline\delta,\kappa$ are positive and nonincreasing.
\end{definition}

At a singular time $t$, surgery is performed using the strong
$\overline\delta(t)$-neck at scale $h(t)$: every boundary sphere is centered at a
point of scalar curvature $h(t)^{-2}$.

\begin{definition}[Time steps $T_i=2^it_0$]
\label{def:time-steps-T-i}
Set $t_0=2^{-5}$ and $T_i=2^it_0$ for $i\geq0$.
\end{definition}

\begin{definition}[Surgery parameter sequences $(r,\kappa,\delta)$]
\label{def:surgery-parameter-sequences}
\uses{thm:local-surgery, prop:neck-gluing, claim:normalized-initial-existence, lem:standard-initial-metric-asymptotic-bounds}

\label{surgeryparams}
Surgery parameter sequences are positive nonincreasing sequences
\[
  {\bf r}:r_0\geq r_1\geq\cdots>0,
  \qquad
  {\bf K}:\kappa_0\geq\kappa_1\geq\cdots>0,
  \qquad
  \Delta:\delta_0\geq\delta_1\geq\cdots>0,
\]
with $r_0=\epsilon$, $\kappa_0$ from Claim~\ref{newkappa0r0t0}, and
\[
  \delta_0=\min(\beta\epsilon/3,\delta'_0,K^{-1},D^{-1}).
\]
Finite initial segments are also called parameter sequences when they satisfy
these conditions.
\end{definition}

Let $r(t)=r_0$ on $[0,T_0)$ and $r(t)=r_{i+1}$ on
$[T_i,T_{i+1})$; define $\kappa(t)$ analogously. A flow has the strong
$(C,\epsilon)$-canonical-neighborhood assumption with parameter ${\bf r}$ if
$R(x)\geq r(t)^{-2}$ implies a strong $(C,\epsilon)$-canonical neighborhood.
It is $\kappa$-noncollapsed on scales $\leq\epsilon$ if every point outside a
positively curved component satisfies the usual parabolic curvature hypothesis
and $\Vol B(x,t,r)\geq\kappa r^3$ for $r\leq\epsilon$. It is ${\bf K}$-noncollapsed
if the restriction to $[0,t]$ has constant $\kappa(t)$ for every $t$. We write
$\overline\delta(t)\leq\Delta$ when
$\overline\delta(t)\leq\delta_0$ on $[0,T_0)$ and
$\overline\delta(t)\leq\delta_{i+1}$ on $[T_i,T_{i+1})$.

\section{The process of surgery}
\label{sect:process}

Given parameter sequences through index $i+1$ and a flow singular at
$T\in(T_i,T_{i+1}]$, take the maximal pre-surgery extension to $T$ and set
$\rho=\overline\delta(T)r_{i+1}$ and
$h(T)=h(\rho,\overline\delta(T))$. The finitely many relevant components of
the limiting region contain disjoint $2\epsilon$-horns. Cut each horn at a
strong $\delta(T)$-neck centered where $R=h(T)^{-2}$, remove its unbounded end,
and glue in a scaled standard cap. Remove components containing no point of
the controlled region and restart the Ricci flow from the resulting compact
metric. The continuing region embeds isometrically, while the exposed region is
a finite union of $3$-balls.

\begin{definition}[The surgery operation at a singular time]
\label{def:surgery-operation}
\uses{thm:local-surgery, lem:gluing-ricci-flows-with-surgery, thm:singular-time-limit-properties, thm:deep-delta-necks-in-horn, lem:finitely-many-horns-meeting-omega-rho}

The preceding cut-and-cap construction at time $T$, using
$\overline\delta(T)$ and $r_{i+1}$, is the surgery operation at $T$.
\end{definition}

\section{Statements about existence}

\begin{theorem}[Main existence theorem: controlled Ricci flow with surgery extends for all time]
\label{thm:main-existence-ricci-flow-surgery}
\uses{def:curvature-pinched-toward-positive, def:strong-canonical-neighborhood-assumption}
\label{MAIN}

There are parameter sequences
\[
  {\bf K}=\{\kappa_i\},\qquad
  \Delta=\{\delta_i\},\qquad
  {\bf r}=\{r_i\}
\]
such that, for the step function $r(t)=r_i$ on each prescribed interval, every
nonincreasing positive $\overline\delta(t)\leq\Delta$ has the following
property. Any Ricci flow with surgery on $[0,T)$ satisfying Assumptions
(1)--(7), positive pinching, the strong $(C,\epsilon)$-canonical-neighborhood
assumption with parameter ${\bf r}$, and ${\bf K}$-noncollapsing extends to a
flow on $[0,\infty)$ satisfying all the same conditions.
\end{theorem}

\begin{proof}
\sketch
Start with the short-time constants of Claim~\ref{newkappa0r0t0}. Inductively
choose $r_{i+1}$, $\kappa_{i+1}$, and $\delta_{i+1}$ so that noncollapsing,
canonical neighborhoods, and surgery control extend over
$[T_i,T_{i+1})$. Lemma~\ref{14.26} preserves the standing assumptions and
pinching after each surgery. The surgery operation and
Lemma~\ref{gluingsoln} restart the flow. On every finite interval, volume grows
at most exponentially while each surgery removes a fixed positive amount, so
only finitely many surgeries occur; induction therefore gives an all-time
flow.
\end{proof}

\begin{corollary}[Every normalized compact 3-manifold admits a Ricci flow with surgery for all time]
\label{cor:ricci-flow-surgery-exists-all-time}
\label{RFSexists}
Let ${\bf K},{\bf r},\Delta$ be supplied by Theorem~\ref{MAIN}, and let
$\overline\delta(t)\leq\Delta$ be positive and nonincreasing. If $M$ is a
compact $3$-manifold with no $\Ar P^2$ having trivial normal bundle, then some
normalized metric $g(0)$ on $M$ is the initial metric of an all-time Ricci flow
with surgery satisfying Assumptions (1)--(7), positive pinching,
${\bf K}$-noncollapsing, and the strong canonical-neighborhood assumption.
Surgeries in $[T_i,T_{i+1})$ use $\overline\delta(t)$ and $r_{i+1}$.
\end{corollary}

\begin{proof}
\sketch
\uses{thm:main-existence-ricci-flow-surgery,prop:normalized-flow-short-time-bounds}
Scale any metric on $M$ to make it normalized. Claim~\ref{newkappa0r0t0}
provides the initial interval and constants; Theorem~\ref{MAIN} extends it
inductively for all time. The hypothesis on $\Ar P^2$ gives Assumption (1).
\end{proof}

\begin{lemma}[Surgery preserves Assumptions (1)--(7) and curvature pinching]
\label{lem:surgery-preserves-assumptions-pinching}
\label{14.26}
Suppose a flow singular at $T\in[T_{i-1},T_i)$ satisfies Assumptions (1)--(7),
positive pinching, the strong canonical-neighborhood assumption, and
${\bf K}$-noncollapsing. For sufficiently small surgery control, the operation
at $T$ yields a flow on $[0,T')$, $T'>T$, satisfying the same assumptions and
positive pinching.
\end{lemma}

\begin{proof}
\sketch
\uses{thm:local-surgery,lem:gluing-ricci-flows-with-surgery,thm:pinching-persists-from-time-a}
The gluing lemma gives the new surgery space-time. Assumptions (1)--(5) follow
from the construction. Points in the disappearing region either approach
infinite curvature or lie in the cut horns or uncontrolled components, hence
have canonical neighborhoods for times sufficiently near $T$; this proves (6).
The restarted flow is maximal, giving (7). The local surgery theorem supplies
pinching at time $T$, and the pinching persistence theorem supplies it for the
new interval.
\end{proof}

\begin{proposition}[The pre-surgery flow map is distance decreasing near a surgery time]
\label{prop:surgery-distance-decreasing}
\uses{thm:local-surgery, def:surgery-parameter-sequences}

\label{surgerydist}
Let $T$ be a surgery time with control below $\delta_0$ and scale below
$R_0^{-1/2}$. For $t'<T$ sufficiently close, the flow gives a compatible
embedding of $M_{t'}\times[t',T)$. On the open set $\Omega$ where the metric
has a limit, this extends isometrically to an open subset of the post-surgery
slice. If every surgery sphere is separating, it extends to a map from each
component $X(t')$ to its post-surgery component $X(T)$, and for $t<T$ near $T$
the extension is distance nonincreasing on
$(X(t')\setminus\Omega,g(t))$.
\end{proposition}

\begin{proof}
\sketch
\uses{thm:local-surgery}
The pre-surgery metric agrees with the limiting metric on $\Omega$ and the
local surgery map is distance decreasing on each capped neck. Separating
spheres allow the local maps to extend over the whole component.
\end{proof}

\begin{remark}[Non-separating surgery spheres prevent extending the comparison map]
\label{rem:nonseparating-surgery-sphere}
\uses{prop:surgery-distance-decreasing}

A nonseparating surgery sphere gives a post-surgery component with caps on both
sides, so the comparison map cannot extend continuously over the entire
pre-surgery component.
\end{remark}
