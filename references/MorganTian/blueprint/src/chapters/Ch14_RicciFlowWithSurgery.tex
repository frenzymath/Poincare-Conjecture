\label{chap:ricci-flow-with-surgery}

\section{Surgery space-time}

\begin{definition}[Space-time: paracompact Hausdorff space with a time function]
\label{def:space-time-general}
A space-time is a paracompact Hausdorff space ${\mathcal M}$ with a continuous
time function ${\bf t}:{\mathcal M}\to\Ar$ whose image is an interval $I$.
The level sets $M_t={\bf t}^{-1}(t)$ are the time-slices; the endpoints of $I$,
when present, give the initial and final time-slices.
\end{definition}

\subsection{The surgery chart}

Let $(-1,1)^2$ carry the topology ${\mathcal P}$ generated by its usual open
sets and by open subsets of $(0,1)\times[0,1)$. Let
$\iota:{\mathcal P}\to(-1,1)^2$ be the identity map. Away from the closed
boundary edge $[0,1)\times\{0\}$ this is the usual manifold topology; away
from the origin, ${\mathcal P}$ is a manifold with boundary. A function is
smooth near $(0,0)$ exactly when it is the pullback by $\iota$ of an ordinary
smooth function near the origin, and at $(a,0)$ with $0<a<1$ exactly when it is
the pullback of a smooth function on an open subset of the upper half-plane.
These prescriptions define a sheaf of smooth functions.

The tangent space is the derivation tangent space of this sheaf. The map
$\iota$ identifies $T{\mathcal P}$ with the pullback of the ordinary tangent
bundle, and a section is smooth precisely when it is
$\alpha\partial_x+\beta\partial_y$ with $\alpha,\beta$ smooth on
${\mathcal P}$. Pull back the second coordinate and its unit vector field to
obtain ${\bf t}_{\mathcal P}$ and $\chi_{\mathcal P}$ with
$\chi_{\mathcal P}({\bf t}_{\mathcal P})=1$; smooth vector fields have unique
local integral curves (one-sided at manifold-with-boundary points).

\subsection{Coordinate charts}

A surgery space-time chart is one of:
\begin{enumerate}
\item an open subset of $\Ar^n\times I$ with ordinary smooth structure,
${\bf t}$ the projection, and $\chi=\partial_t$;
\item an open subset of $\Ar^n\times[a,\infty)$ with its manifold-with-boundary
structure, projection time, and $\chi=\partial_t$, where $a$ is not the initial
time;
\item a product of an open subset of $\Ar^{n-1}$ with ${\mathcal P}$, with the
product smooth structure, time equal up to translation to
${\bf t}_{\mathcal P}$, and $\chi$ induced by $\chi_{\mathcal P}$.
\end{enumerate}
Overlap maps are diffeomorphisms in the corresponding smooth structures and
preserve ${\bf t}$ and $\chi$. Charts of types 1 and 2 meet type 3 only away
from its exotic boundary. Type 1 charts describe ordinary Ricci flows; type 2
charts are noninitial manifold-with-boundary regions.

\begin{definition}[Surgery space-time: smooth, exposed, and singular points]
\label{def:surgery-space-time}
\uses{def:space-time-general}

\label{spacetime}
A surgery space-time is a space-time with a maximal atlas of these three chart
types and the stated overlap maps. Points with type 1 neighborhoods are
\emph{smooth}; points with type 2 but no type 1 neighborhood are
\emph{exposed}; all remaining points are \emph{singular}. Smooth and exposed
points form a smooth manifold with boundary. Its boundary components lie in
time-slices; those not in the initial or final slice form the exposed region,
and the boundary of the exposed region is the singular set.

The local time functions and vector fields glue to global smooth objects
${\bf t}:{\mathcal M}\to\Ar$ and $\chi$ with
\[
  \chi({\bf t})=1.
\]
The tangent bundles glue as well. The vector field points into the space-time
along exposed and initial boundaries and out along the final boundary.
\end{definition}

\begin{definition}[Embedding compatible with time and the vector field; maximal flow line]
\label{def:compatible-with-time-vector-field}
\uses{def:surgery-space-time}

An embedding $K\times J\to{\mathcal M}$ is compatible with time and $\chi$ if
${\bf t}$ restricts to the projection on $J$ and each
$\{x\}\times J$ maps to the $\chi$-integral curve through $x$. If
$K\subset M_t$, require that $t\in J$ and the slice map at $t$ is the identity.
Uniqueness of integral curves gives a maximal such interval $J_K$; for a point
$x$, its embedding is the maximal flow line and its interval is the domain of
definition of that flow line. A compatible embedding $K\times J$ exists exactly
when $J$ lies in the domain of every flow line through $K$.
\end{definition}

\begin{definition}[Regular and singular times of a surgery space-time]
\label{def:regular-singular-time-surgery}
\uses{def:surgery-space-time, def:compatible-with-time-vector-field}

A time $t$ is regular if some neighborhood $J\subset I$ of $t$ admits a
compatible diffeomorphism $M_t\times J\to{\bf t}^{-1}(J)$. Otherwise it is
singular. If every time is regular, the space-time is the product
$M_t\times I$ with product time and vector field.
\end{definition}

\begin{lemma}[Time-slices are smooth manifolds with a complementary horizontal subbundle]
\label{lem:time-slice-smooth-manifold}
\uses{def:surgery-space-time}

Every time-slice $M_t$ is a smooth $n$-manifold. Its tangent bundle embeds as a
codimension-one subbundle of $T{\mathcal M}|_{M_t}$ complementary to the line
spanned by $\chi$; these subbundles vary smoothly and form a global horizontal
subbundle.
\end{lemma}

\begin{proof}
\sketch
In each chart, $M_t$ is an ordinary coordinate slice and its tangent space is
the kernel of $d{\bf t}$, complementary to $\chi$ because
$\chi({\bf t})=1$. The local descriptions agree under the overlap maps.
\end{proof}

\begin{definition}[Horizontal subbundle of a surgery space-time]
\label{def:horizontal-subbundle}
\uses{lem:time-slice-smooth-manifold}

The codimension-one bundle from the preceding lemma is the horizontal
subbundle and is denoted ${\mathcal HT}({\mathcal M})$.
\end{definition}

\section{The generalized Ricci flow equation}

\begin{definition}[Horizontal metric and its curvatures on a surgery space-time]
\label{def:horizontal-metric}
\uses{def:horizontal-subbundle}

A horizontal metric $G$ is a smooth metric on
${\mathcal HT}({\mathcal M})$. Its restriction $G(t)$ to $M_t$ has the usual
Levi--Civita connection, gradient, curvature operator $\Rm_G$, Ricci tensor
$\Ric_G$, and scalar curvature $R_G$. These horizontal tensors vary smoothly
over the surgery space-time.
\end{definition}

\begin{definition}[Ricci flow with surgery: the generalized Ricci flow equation]
\label{def:ricci-flow-with-surgery}
\uses{def:horizontal-metric, def:surgery-space-time}

A Ricci flow with surgery is a pair $({\mathcal M},G)$ consisting of a surgery
space-time and horizontal metric satisfying
\begin{equation}
\label{GRFEq}
  {\mathcal L}_\chi G=-2\Ric_G
\end{equation}
as sections of the symmetric square of the horizontal cotangent bundle. If
${\mathcal M}$ has dimension $n+1$, the flow is $n$-dimensional.
\end{definition}

\begin{remark}[Lie derivative is one-sided at exposed points and initial/final times]
\label{rem:one-sided-lie-derivative}
\uses{def:ricci-flow-with-surgery}

At exposed points and at initial or final time-slices,
${\mathcal L}_\chi$ is the corresponding one-sided derivative.
\end{remark}

\begin{example}[An ordinary Ricci flow is a Ricci flow with surgery]
\label{ex:ordinary-flow-as-surgery-flow}
\uses{def:ricci-flow-with-surgery}

For ${\mathcal M}=M_0\times[0,T)$ with product time and vector field,
${\mathcal L}_\chi G=\partial_tg(t)$, so Equation~\eqref{GRFEq} is the
ordinary Ricci flow equation.
\end{example}

\begin{lemma}[Gluing two Ricci flows along a common time-slice yields a Ricci flow with surgery]
\label{lem:gluing-ricci-flows-with-surgery}
\uses{def:ricci-flow-with-surgery}

\label{gluingsoln}
Let $M_1\times(a,b]$ and $M_2\times[b,c)$ carry compact codimension-zero
submanifolds $\Omega_i$ with open neighborhoods $U_i$, and let
$\psi:U_1\to U_2$ be a diffeomorphism carrying $\Omega_1$ to $\Omega_2$.
Glue $\Omega_1\times\{b\}$ to $\Omega_2\times\{b\}$ by $\psi$, deleting
$(M_1\setminus\Omega_1)\times\{b\}$ from the first piece. The resulting
space has a surgery space-time structure. If Ricci flows $g_1(t)$ and $g_2(t)$
fit at $b$ under an isometry
$\psi:(U_1,g_1(b))\to(U_2,g_2(b))$, their horizontal metrics glue to a Ricci
flow with surgery.
\end{lemma}

\begin{proof}
\sketch
\uses{prop:smooth-patching}
Away from the gluing boundary use the ordinary, manifold-with-boundary, or
exotic chart according to the three cases. Near
$\partial\Omega_1\times\{b\}$ the exotic half-square topology gives type 3
coordinates. The time and vector field glue by construction. The isometry at
$b$ and smooth patching of Ricci flows make the horizontal metric smooth and
preserve Equation~\eqref{GRFEq}.
\end{proof}

\begin{proposition}[Smooth and exposed points of a Ricci flow with surgery form a generalized Ricci flow]
\label{prop:interior-generalized-ricci-flow}
\uses{def:ricci-flow-with-surgery, def:generalized-ricci-flow}

For a Ricci flow with surgery, the open subset consisting of smooth points and
manifold-with-boundary points at the initial or final time inherits a smooth
manifold-with-boundary structure. Restricting ${\bf t}$, $\chi$, and $G$ to it
gives a generalized Ricci flow.
\end{proposition}

\subsection{Scaling and translation}

If $({\mathcal M},G)$ is a Ricci flow with surgery and $Q>0$, then
\[
  G'=QG,\qquad {\bf t}'=Q{\bf t},\qquad \chi'=Q^{-1}\chi
\]
defines another Ricci flow with surgery, denoted $(Q{\mathcal M},QG)$. Replacing
${\bf t}$ by ${\bf t}+a$ gives time translation without changing $G$ or $\chi$.

\begin{definition}[Metric ball in a time-slice of space-time]
\label{def:metric-ball-spacetime}
\uses{def:surgery-space-time}

For $x\in{\mathcal M}$, $t={\bf t}(x)$, and $r>0$,
$B(x,t,r)$ is the radius-$r$ metric ball about $x$ in $(M_t,G(t))$.
\end{definition}

\begin{definition}[Forward and backward parabolic neighborhoods in a Ricci flow with surgery]
\label{def:parabolic-neighborhood-surgery}
\uses{def:metric-ball-spacetime, def:compatible-with-time-vector-field}

The backward neighborhood $P(x,t,r,-\Delta t)$ exists if there is a compatible
embedding $B(x,t,r)\times(t-\Delta t,t]\to{\mathcal M}$. The forward
neighborhood $P(x,t,r,\Delta t)$ exists if there is a compatible embedding
$B(x,t,r)\times[t,t+\Delta t)\to{\mathcal M}$. Either is a parabolic
neighborhood.
\end{definition}

\begin{definition}[Kappa-non-collapsed on scales at most $r_0$, for surgery flows]
\label{def:kappa-noncollapsed-surgery}
\uses{def:parabolic-neighborhood-surgery}

\label{defnkappa}
For $\kappa,r_0>0$, a Ricci flow with surgery is
$\kappa$-noncollapsed on scales $\leq r_0$ if, whenever $r\leq r_0$,
$P(x,t,r,-r^2)$ exists and $|\Rm_G|\leq r^{-2}$ there, then
\[
  \Vol B(x,t,r)\geq\kappa r^3.
\]
\end{definition}

\begin{remark}[Non-collapsing constant for epsilon-round components depends on fundamental group order]
\label{rem:epsilon-round-kappa-dependence}
\uses{def:kappa-noncollapsed-surgery, def:epsilon-round-component}

For sufficiently small $\epsilon$, an $\epsilon$-round component satisfies the
preceding condition with a constant depending on the order of its finite
fundamental group. If every finite free factor of the initial fundamental group
has order at most $N$, one may choose $\kappa$ depending only on $N$.
\end{remark}

\begin{definition}[Curvature pinched toward positive, for a Ricci flow with surgery]
\label{def:curvature-pinched-toward-positive-surgery}
\uses{def:ricci-flow-with-surgery, def:curvature-operator}

For a $3$-dimensional flow with time in $[0,\infty)$, let
$\lambda\geq\mu\geq\nu$ be the eigenvalues of $\Rm$ and set
$X=\max(0,-\nu)$. Curvature is \emph{pinched toward positive} if
\[
  R(x)\geq-\frac6{1+4{\bf t}(x)}
\]
and, when $X(x)>0$,
\[
  R(x)\geq2X(x)\bigl(\log X(x)+\log(1+{\bf t}(x))-3\bigr).
\]
For a metric $(M,g)$ and $T\geq0$, pinching up to time $T$ means the same
inequalities with ${\bf t}(x)$ replaced by $T$.
\end{definition}

\begin{definition}[Strong (C,epsilon)-canonical neighborhood assumption with parameter r]
\label{def:strong-canonical-neighborhood-assumption}
\uses{def:evolving-epsilon-neck, def:canonical-neighborhood, def:strong-epsilon-neck-canonical-neighborhood, def:compatible-with-time-vector-field}

Fix $C,\epsilon,r>0$. A Ricci flow with surgery satisfies the strong
$(C,\epsilon)$-canonical-neighborhood assumption with parameter $r$ if every
point with $R\geq r^{-2}$ has a strong $(C,\epsilon)$-canonical neighborhood.
For cap, round-component, and ordinary neck cases this is the corresponding
time-slice notion. For a strong $\epsilon$-neck centered at $x$, it requires a
compatible embedding
\[
  (S^2\times(-\epsilon^{-1},\epsilon^{-1}))
  \times({\bf t}(x)-R(x)^{-1},{\bf t}(x)]
  \longrightarrow{\mathcal M}
\]
whose rescaling by $R(x)$ and time translation by $-{\bf t}(x)$ is
$\epsilon$-close in $C^{[1/\epsilon]}$ to the evolving round cylinder with
scalar curvature $1-t$, $-1<t\leq0$. An exposed point cannot be the center of
a strong $\epsilon$-neck.
\end{definition}
