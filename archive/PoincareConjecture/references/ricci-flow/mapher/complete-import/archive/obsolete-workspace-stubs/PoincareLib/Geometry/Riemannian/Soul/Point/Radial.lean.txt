/-!
# Distance-parametrized radial coordinates

The radial flow is reparametrized by its actual intrinsic distance.  The
result is a homeomorphism of the unit two-sphere times positive radii with the
punctured manifold and records the distance identity used by annulus
consumers.
-/

noncomputable section
set_option autoImplicit false

open Set
open scoped Manifold ContDiff Bundle ENNReal Topology

namespace PoincareMT.RiemannianMetric

variable {M : Type*} [TopologicalSpace M] [T3Space M]
  [ConnectedSpace M] [Nonempty M]
  [ChartedSpace (EuclideanSpace ℝ (Fin 3)) M]
  [IsManifold (𝓡 3) ∞ M] [NoncompactSpace M]

/-- The distance-preserving radial homeomorphism associated to a point soul. -/
theorem exists_radialHomeomorph_of_point_soul
    (g : RiemannianMetric 3 M) (D : LeviCivitaData g)
    (hcomplete : MetricComplete g)
    (hpos : D.StrictlyPositiveSectionalCurvature)
    {p : M} (hp : TotallyConvexSet g ({p} : Set M)) :
    Nonempty (RadialHomeomorph g p) := by
  sorry

end PoincareMT.RiemannianMetric
